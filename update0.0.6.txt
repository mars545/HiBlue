class MainActivity : Activity() {
    private companion object {
        const val ASSET_PREFIX = "file:///android_asset/"
        const val START_URL = "file:///android_asset/index.html"
        const val REQ_NOTIF = 7
        const val REQ_NATIVE_GAME = 9
    }

    private lateinit var wv: WebView
    private lateinit var device: DeviceServices
    private var lastBack = 0L
    private var insTop = 0
    private var insBottom = 0
    @Volatile private var pendingNativeId: String? = null

    /** Setat doar din firul UI (onPageStarted); citit din firul JavaBridge. */
    @Volatile private var pageTrusted = false

    private val actions = object : HostActions {
        override fun trustedPage(): Boolean = pageTrusted

        override fun vibrate(kind: Int) {
            runOnUiThread { device.vibrate(kind) }
        }

        override fun notificationsGranted(): Boolean = device.notificationsGranted()

        override fun requestNotifications() {
            runOnUiThread {
                if (Build.VERSION.SDK_INT >= 33 && !device.notificationsGranted()) {
                    requestPermissions(arrayOf(Manifest.permission.POST_NOTIFICATIONS), REQ_NOTIF)
                }
            }
        }

        override fun notify(title: String, body: String): Boolean = device.notify(title, body)

        override fun setOrientation(mode: String): Boolean {
            val v = when (mode) {
                "portrait" -> ActivityInfo.SCREEN_ORIENTATION_PORTRAIT
                "landscape" -> ActivityInfo.SCREEN_ORIENTATION_SENSOR_LANDSCAPE
                "auto" -> ActivityInfo.SCREEN_ORIENTATION_FULL_SENSOR
                else -> return false
            }
            runOnUiThread { requestedOrientation = v }
            return true
        }

        override fun nativeGames(): List<Pair<String, String>> =
            NativeGames.list().map { Pair(it.id, it.runtime) }

        override fun launchNativeGame(id: String): String? {
            val entry = NativeGames.get(id) ?: return "NOT_REGISTERED"
            if (pendingNativeId != null) return "BUSY"
            pendingNativeId = id
            runOnUiThread {
                try {
                    val i = entry.intentFactory(this@MainActivity)
                    i.putExtra(NativeGames.EXTRA_GAME_ID, id)
                    @Suppress("DEPRECATION")
                    startActivityForResult(i, REQ_NATIVE_GAME)
                } catch (e: Exception) {
                    pendingNativeId = null
                    emit("nativeGame", JSONObject().put("id", id).put("status", "error"))
                }
            }
            return null
        }
    }

    override fun onCreate(b: Bundle?) {
        super.onCreate(b)
        NativeGames.registerBuiltIn()
        device = DeviceServices(this)

        val bg = Color.parseColor("#05050F")
        window.statusBarColor = bg
        window.navigationBarColor = bg
        if (Build.VERSION.SDK_INT >= 28) {

            lp.layoutInDisplayCutoutMode = WindowManager.LayoutParams.LAYOUT_IN_DISPLAY_CUTOUT_MODE_SHORT_EDGES
            window.attributes = lp

        WindowCompat.setDecorFitsSystemWindows(window, false)

        val root = FrameLayout(this)

        setContentView(root)

            val bars = ins.getInsets(WindowInsetsCompat.Type.systemBars() or WindowInsetsCompat.Type.displayCutout())

        if (pageTrusted) emit("lifecycle", JSONObject().put("state", "pause"))
        wv.onPause()
        super.onPause()
    }

    override fun onResume() {
        super.onResume()
        wv.onResume()
        if (pageTrusted) emit("lifecycle", JSONObject().put("state", "resume"))
    }

    override fun onRequestPermissionsResult(code: Int, p: Array<out String>, r: IntArray) {
        super.onRequestPermissionsResult(code, p, r)
        if (code == REQ_NOTIF) {
            val ok = r.isNotEmpty() && r[0] == PackageManager.PERMISSION_GRANTED
            emit("permission", JSONObject().put("name", "notifications").put("granted", ok))
        }
    }

    @Suppress("DEPRECATION")
    override fun onActivityResult(code: Int, result: Int, data: Intent?) {
        super.onActivityResult(code, result, data)
        if (code != REQ_NATIVE_GAME) return
        val id = pendingNativeId ?: return
        pendingNativeId = null
        val o = JSONObject().put("id", id)
        if (result == RESULT_OK && data != null) {
            val score = data.getIntExtra(NativeGames.EXTRA_SCORE, -1)
            val secs = data.getIntExtra(NativeGames.EXTRA_SECONDS, 0)
            o.put("status", "finished").put("score", if (score < 0) 0 else score).put("seconds", if (secs < 0) 0 else secs)
        } else {
            o.put("status", "cancelled")
        }
        emit("nativeGame", o)
    }

    override fun onSaveInstanceState(o: Bundle) { super.onSaveInstanceState(o); wv.saveState(o) }

    @Suppress("DEPRECATION")
    override fun onBackPressed() {
        wv.evaluateJavascript("(window.HB&&HB.back)?HB.back():false") { r ->
            if (r == "true") return@evaluateJavascript
            val now = System.currentTimeMillis()
            if (now - lastBack < 2000) finish()
            else { lastBack = now; Toast.makeText(this, "Apasă încă o dată pentru a ieși", Toast.LENGTH_SHORT).show() }
        }
    }

    override fun onDestroy() {
        pageTrusted = false
        try { wv.removeJavascriptInterface(HostBridge.NAME) } catch (e: Exception) {}
        try { (wv.parent as? ViewGroup)?.removeView(wv) } catch (e: Exception) {}
        try { wv.destroy() } catch (e: Exception) {}
        super.onDestroy()
    }
}

package ro.lingurar.hiblue.bridge

import android.webkit.JavascriptInterface
import org.json.JSONArray
import org.json.JSONObject

/**
 * Acțiunile pe care host-ul le pune la dispoziția punții. Implementate de MainActivity.
 * Metodele pot fi apelate din firul JavaBridge (nu din UI); implementarea trebuie să sară pe UI când e nevoie.
 */
interface HostActions {
    /** Pagina încărcată acum în WebView este pagina locală a aplicației (assets)? */
    fun trustedPage(): Boolean
    fun vibrate(kind: Int)
    fun notificationsGranted(): Boolean
    fun requestNotifications()
    fun notify(title: String, body: String): Boolean
    fun setOrientation(mode: String): Boolean
    /** null = pornit, altfel codul erorii */
    fun launchNativeGame(id: String): String?
    /** (id, runtime) */
    fun nativeGames(): List<Pair<String, String>>
}

/**
 * Singura suprafață Kotlin expusă JavaScript-ului: window.HBNative.invoke(json) -> json.
 *
 * Cerere:   {"v":1,"method":"device.vibrate","params":{"kind":1}}
 * Răspuns:  {"v":1,"ok":true,"data":{...}}  sau  {"v":1,"ok":false,"error":{"code":"...","message":"..."}}
 *
 * Reguli: metode pe listă albă, parametri validați strict, cerere limitată ca dimensiune,
 * refuz dacă pagina curentă nu este cea locală. Nu se expun tokenuri, parole sau chei.
 */
class HostBridge(private val actions: HostActions, private val appVersion: String) {

    companion object {
        const val NAME = "HBNative"
        const val PROTOCOL = 1
        private const val MAX_REQUEST = 8 * 1024
        private val CAPABILITIES = listOf(
            "device.vibrate",
            "device.notifications",
            "device.notify",
            "host.orientation",
            "game.native"
        )
        private val GAME_ID = Regex("^[a-z0-9][a-z0-9_-]{1,31}$")
    }

    private fun ok(data: JSONObject): String =
        JSONObject().put("v", PROTOCOL).put("ok", true).put("data", data).toString()

    private fun fail(code: String, message: String): String =
        JSONObject().put("v", PROTOCOL).put("ok", false)
            .put("error", JSONObject().put("code", code).put("message", message)).toString()

    @JavascriptInterface
    fun invoke(request: String?): String {
        if (!actions.trustedPage()) return fail("FORBIDDEN", "Pagină neautorizată")
        if (request == null || request.length > MAX_REQUEST) return fail("BAD_REQUEST", "Cerere invalidă")
        val req: JSONObject = try {
            JSONObject(request)
        } catch (e: Exception) {
            return fail("BAD_REQUEST", "Cerere invalidă")
        }
        val method = req.optString("method", "")
        val params = req.optJSONObject("params") ?: JSONObject()
        return try {
            dispatch(method, params)
        } catch (e: Exception) {
            fail("INTERNAL", "Eroare internă")
        }
    }

    private fun dispatch(method: String, p: JSONObject): String {
        when (method) {
            "host.info" -> {
                return ok(
                    JSONObject().put("protocol", PROTOCOL).put("app", appVersion)
                        .put("capabilities", JSONArray(CAPABILITIES))
                )
            }
            "device.vibrate" -> {
                val k = p.opt("kind")
                if (k !is Int || k < 0 || k > 2) return fail("BAD_PARAMS", "kind trebuie să fie 0, 1 sau 2")
                actions.vibrate(k)
                return ok(JSONObject())
            }
            "device.notifications.status" -> {
                return ok(JSONObject().put("granted", actions.notificationsGranted()))
            }
            "device.notifications.request" -> {
                actions.requestNotifications()
                return ok(JSONObject().put("requested", true))
            }
            "device.notify" -> {
                val t = p.opt("title")
                val b = p.opt("body")
                if (t !is String || b !is String || t.isEmpty() || t.length > 60 || b.length > 200) {
                    return fail("BAD_PARAMS", "title (1-60) și body (0-200) sunt obligatorii")
                }
                if (!actions.notificationsGranted()) return fail("PERMISSION_DENIED", "Notificările nu sunt permise")
                return if (actions.notify(t, b)) ok(JSONObject()) else fail("INTERNAL", "Notificarea nu a putut fi afișată")
            }
            "host.orientation" -> {
                val m = p.opt("mode")
                if (m !is String || (m != "portrait" && m != "landscape" && m != "auto")) {
                    return fail("BAD_PARAMS", "mode: portrait | landscape | auto")
                }
                return if (actions.setOrientation(m)) ok(JSONObject()) else fail("INTERNAL", "Orientare neschimbată")
            }
            "game.native.list" -> {
                val arr = JSONArray()
                for (g in actions.nativeGames()) arr.put(JSONObject().put("id", g.first).put("runtime", g.second))
                return ok(JSONObject().put("games", arr))
            }
            "game.native.launch" -> {
                val id = p.opt("id")
                if (id !is String || !GAME_ID.matches(id)) return fail("BAD_PARAMS", "id de joc invalid")
                val err = actions.launchNativeGame(id)
                return if (err == null) ok(JSONObject().put("launched", true)) else fail(err, "Jocul nativ nu a putut porni")
            }
            else -> return fail("UNKNOWN_METHOD", "Metodă necunoscută")
        }
    }
}

package ro.lingurar.hiblue.host

import android.Manifest
import android.app.Notification
import android.app.NotificationChannel
import android.app.NotificationManager
import android.app.PendingIntent
import android.content.Context
import android.content.Intent
import android.content.pm.PackageManager

import android.os.VibrationEffect
import android.os.Vibrator
import ro.lingurar.hiblue.MainActivity

/** Funcții native ale dispozitivului: vibrații și notificări. Apelate doar prin HostBridge. */
class DeviceServices(private val ctx: Context) {

    private companion object {
        const val CHANNEL = "hiblue_default"
    }

    @Suppress("DEPRECATION")
    fun vibrate(kind: Int) {
        try {
            val v = ctx.getSystemService(Context.VIBRATOR_SERVICE) as Vibrator
            if (!v.hasVibrator()) return
            if (Build.VERSION.SDK_INT >= 29) {
                val e = when (kind) {
                    0 -> VibrationEffect.EFFECT_TICK
                    1 -> VibrationEffect.EFFECT_CLICK
                    else -> VibrationEffect.EFFECT_HEAVY_CLICK
                }
                v.vibrate(VibrationEffect.createPredefined(e))
            } else {
                val ms = when (kind) {
                    0 -> 12L
                    1 -> 20L
                    else -> 32L
                }
                if (Build.VERSION.SDK_INT >= 26) {
                    v.vibrate(VibrationEffect.createOneShot(ms, VibrationEffect.DEFAULT_AMPLITUDE))
                } else {
                    v.vibrate(ms)
                }
            }
        } catch (e: Exception) {
        }
    }

    fun notificationsGranted(): Boolean =
        Build.VERSION.SDK_INT < 33 ||
            ctx.checkSelfPermission(Manifest.permission.POST_NOTIFICATIONS) == PackageManager.PERMISSION_GRANTED

    @Suppress("DEPRECATION")
    fun notify(title: String, body: String): Boolean {
        if (!notificationsGranted()) return false
        return try {
            val nm = ctx.getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
            val builder: Notification.Builder
            if (Build.VERSION.SDK_INT >= 26) {
                nm.createNotificationChannel(NotificationChannel(CHANNEL, "HiBlue", NotificationManager.IMPORTANCE_DEFAULT))
                builder = Notification.Builder(ctx, CHANNEL)
            } else {
                builder = Notification.Builder(ctx)
            }
            val open = PendingIntent.getActivity(
                ctx, 0,
                Intent(ctx, MainActivity::class.java).addFlags(Intent.FLAG_ACTIVITY_SINGLE_TOP),
                PendingIntent.FLAG_IMMUTABLE or PendingIntent.FLAG_UPDATE_CURRENT
            )
            builder.setSmallIcon(android.R.drawable.ic_dialog_info)
                .setContentTitle(title)
                .setContentText(body)
                .setAutoCancel(true)
                .setContentIntent(open)
            nm.notify(1001, builder.build())
            true
        } catch (e: Exception) {
            false
        }
    }
}

/* =====================================================================
   HiBlue · NativeBridge (client JS al punții către Android)
   Singurul loc din frontend care vorbește cu partea nativă. Restul codului
   (UI, jocuri, SDK) apelează NativeBridge.*, nu window.HBNative.

   Protocol v1 (JSON, vezi bridge/HostBridge.kt):
     cerere  {v:1, method, params}
     răspuns {v:1, ok:true, data} | {v:1, ok:false, error:{code,message}}
   Toate metodele întorc {ok,data|error}; NU aruncă excepții.
   În browser / fără Android: funcțiile indisponibile întorc {ok:false, error:{code:"UNAVAILABLE"}}
   iar vibrate() cade pe navigator.vibrate când există.

   Evenimente Kotlin -> JS:  NativeBridge.on("lifecycle"|"permission"|"nativeGame", fn)
   ===================================================================== */
(function (root) {
  "use strict";
  var PROTOCOL = 1;
  var listeners = {};
  var cachedInfo = null;

  function raw() {
    try {
      var n = root.HBNative;
      return n && typeof n.invoke === "function" ? n : null;
    } catch (e) { return null; }
  }
  function fail(code, message) { return { ok: false, error: { code: code, message: message || code } }; }

  function call(method, params) {
    var n = raw();
    if (!n) return fail("UNAVAILABLE", "Funcția nativă nu este disponibilă");
    if (typeof method !== "string" || !method) return fail("BAD_PARAMS", "method lipsește");
    var out;
    try {
      out = JSON.parse(n.invoke(JSON.stringify({ v: PROTOCOL, method: method, params: params || {} })));
    } catch (e) {
      return fail("INTERNAL", String(e && e.message || e));
    }
    if (out && out.ok === true) return { ok: true, data: out.data || {} };
    var er = out && out.error || {};
    return fail(er.code || "INTERNAL", er.message);
  }

  function info() {
    if (cachedInfo) return cachedInfo;
    var r = call("host.info");
    if (r.ok) cachedInfo = r;
    return r;
  }
  function has(cap) {
    var i = info();
    return !!(i.ok && Array.isArray(i.data.capabilities) && i.data.capabilities.indexOf(cap) >= 0);
  }

  function vibrate(kind) {
    var k = kind === 1 || kind === 2 ? kind : 0;
    if (raw()) return call("device.vibrate", { kind: k }).ok;
    try {
      if (root.navigator && root.navigator.vibrate) return !!root.navigator.vibrate(k === 2 ? 28 : k === 1 ? 18 : 10);
    } catch (e) { /* ignorat */ }
    return false;
  }

  var notifications = {
    granted: function () {
      var r = call("device.notifications.status");
      return r.ok ? !!r.data.granted : false;
    },
    request: function () { return call("device.notifications.request"); },

# HiBlue · Android Host și NativeBridge (update0.0.6)

## Rolul Android Host
Android (Kotlin) este stratul nativ de bază. NU construiește interfețe: interfața rămâne în `assets/` (HTML/CSS/JS).

| Responsabilitate | Fișier |
|---|---|
| Lifecycle, WebView, inset-uri, Back, permisiuni | `MainActivity.kt` |
| Puntea JS → Kotlin (listă albă de metode, validare) | `bridge/HostBridge.kt` |
| Vibrații, notificări | `host/DeviceServices.kt` |
| Registru jocuri native + contract de rezultat | `nativegames/NativeGames.kt` |
| Client JS al punții (singurul loc care vorbește cu Android) | `assets/js/native-bridge.js` |

## Protocol (v1)
JS apelează `window.HBNative.invoke(json)` (doar prin `NativeBridge.call`). Cerere:
`{"v":1,"method":"device.vibrate","params":{"kind":1}}`. Răspuns:
`{"v":1,"ok":true,"data":{}}` sau `{"v":1,"ok":false,"error":{"code":"BAD_PARAMS","message":"..."}}`.

Metode: `host.info`, `device.vibrate {kind 0..2}`, `device.notifications.status`,
`device.notifications.request`, `device.notify {title 1-60, body 0-200}`,
`host.orientation {mode: portrait|landscape|auto}`, `game.native.list`, `game.native.launch {id}`.

Coduri de eroare: `FORBIDDEN`, `BAD_REQUEST`, `BAD_PARAMS`, `UNKNOWN_METHOD`, `PERMISSION_DENIED`,
`NOT_REGISTERED`, `BUSY`, `INTERNAL`, `UNAVAILABLE` (doar client, fără Android).

Evenimente Kotlin → JS (`NativeBridge.on(tip, fn)`): `lifecycle {state: pause|resume}`,
`permission {name, granted}`, `nativeGame {id, status: finished|cancelled|error, score, seconds}`.

## Securitate
- Se expune o singură metodă (`invoke`); nu există tokenuri, parole sau chei în punte.
- Puntea refuză orice apel dacă pagina curentă nu este `file:///android_asset/...` (`FORBIDDEN`).
- Navigarea în WebView către orice în afara assets este blocată; linkurile http(s)/mailto/tel se deschid în browserul sistemului.
- `frame-src 'none'` (CSP): nicio pagină externă nu poate fi încadrată într-un iframe cu acces la punte.
- WebView: fără acces la fișiere externe, fără geolocație, fără ferestre multiple, fără conținut mixt; debugging doar în build debuggable.

## De ce NU s-a folosit WebViewAssetLoader (încă)
`WebViewAssetLoader` servește assets de pe `https://appassets.androidplatform.net`. Asta schimbă *originea* paginii
față de `file://` folosită de instalările existente, iar `localStorage` este per-origine: sesiunea Supabase,
setările, recordurile și salvarea My Blue (`hiblue.myblue.save.v1`) nu ar mai fi vizibile după update.
Cerința de compatibilitate cu instalările existente are prioritate. Migrarea se poate face într-o etapă separată
(o pagină `file://` care exportă `localStorage` prin punte către noua origine, o singură dată).

## Jocuri native
Vezi `docs/NATIVE_GAMES.md` (creat în update0.0.7). În 0.0.6 există doar infrastructura din host:
registrul este gol, deci `game.native.launch` întoarce `NOT_REGISTERED`.
