    override fun onSurfaceChanged(gl: GL10?, w: Int, h: Int) {
        viewportWidth = max(1, w); viewportHeight = max(1, h)

    override fun onDrawFrame(gl: GL10?) {

        val dt = (now-lastFrame) / 1_000_000_000f
        lastFrame = now
        world.update(dt)
        world.camera(camera)
        GLES30.glClear(GLES30.GL_COLOR_BUFFER_BIT or GLES30.GL_DEPTH_BUFFER_BIT)
        val targetX = camera[0] + sin(camera[3]) * cos(camera[4])
        val targetY = camera[1] + sin(camera[4])
        val targetZ = camera[2] + cos(camera[3]) * cos(camera[4])
        Matrix.setLookAtM(view,0,camera[0],camera[1],camera[2],targetX,targetY,targetZ,0f,1f,0f)
        Matrix.multiplyMM(vp,0,projection,0,view,0)
        GLES30.glUseProgram(program)
        GLES30.glUniform1f(uTime, world.dayTime)
        draw(water, identity)
        draw(ground, identity)

        drawScaled(sphere, 0f, 39f, -118f, 10f,10f,10f)

        }
        for (p in plants) {
            if (distanceSq(p.x,p.z,camera[0],camera[2]) > 160f*160f) continue
            if (p.type == 0) {
                val sway = sin(world.elapsed*1.35f+p.seed)*0.05f
                drawScaled(cylinder,p.x, p.y+0.72f*p.scale,p.z,0.13f*p.scale,0.76f*p.scale,0.13f*p.scale, colorOverride=floatArrayOf(0.35f,0.25f,0.17f,1f))
                drawScaled(sphere,p.x+sway,p.y+1.35f*p.scale,p.z,0.82f*p.scale,0.88f*p.scale,0.77f*p.scale, colorOverride=floatArrayOf(0.25f,0.48f,0.31f,1f))
                drawScaled(sphere,p.x-sway*1.2f,p.y+1.75f*p.scale,p.z+0.06f,0.63f*p.scale,0.61f*p.scale,0.63f*p.scale, colorOverride=floatArrayOf(0.34f,0.58f,0.37f,1f))
                drawScaled(sphere,p.x,p.y+1.17f*p.scale,p.z,0.52f*p.scale,0.54f*p.scale,0.54f*p.scale, colorOverride=floatArrayOf(0.47f,0.65f,0.39f,1f))

                drawScaled(cone,p.x,p.y+1.25f*p.scale,p.z,0.72f*p.scale,1.2f*p.scale,0.72f*p.scale, colorOverride=floatArrayOf(0.26f,0.49f,0.34f,1f))
                drawScaled(cone,p.x,p.y+1.95f*p.scale,p.z,0.49f*p.scale,0.95f*p.scale,0.49f*p.scale, colorOverride=floatArrayOf(0.34f,0.57f,0.39f,1f))
            }
        }
        landmarks.forEach { l -> drawLandmark(l) }
        world.campfirePositions().forEach { (fireX, fireZ) ->
            if (distanceSq(fireX,fireZ,camera[0],camera[2]) < 155f*155f) {
                val fireY = terrainHeight(fireX,fireZ) + 0.18f

        }

    }

    private fun draw(mesh: Mesh, matrix: FloatArray, override: FloatArray? = null) {
        GLES30.glBindBuffer(GLES30.GL_ARRAY_BUFFER, mesh.vbo)
        val stride = 7 * 4
        GLES30.glEnableVertexAttribArray(aPosition)
        GLES30.glVertexAttribPointer(aPosition,3,GLES30.GL_FLOAT,false,stride,0)
        GLES30.glEnableVertexAttribArray(aColor)
        GLES30.glVertexAttribPointer(aColor,4,GLES30.GL_FLOAT,false,stride,12)
        if (override != null) {
            // Uniform-free tinting is baked into a compact reusable object buffer below.
            GLES30.glDisableVertexAttribArray(aColor)
            GLES30.glVertexAttrib4f(aColor,override[0],override[1],override[2],override[3])
        }
        Matrix.multiplyMM(mvp,0,vp,0,matrix,0)
        GLES30.glUniformMatrix4fv(uMvp,1,false,mvp,0)
        if (mesh.ibo != 0) {
            GLES30.glBindBuffer(GLES30.GL_ELEMENT_ARRAY_BUFFER,mesh.ibo)
            GLES30.glDrawElements(GLES30.GL_TRIANGLES,mesh.count,GLES30.GL_UNSIGNED_SHORT,0)
        } else GLES30.glDrawArrays(GLES30.GL_TRIANGLES,0,mesh.count)
        if (override != null) GLES30.glEnableVertexAttribArray(aColor)
    }

    private fun drawScaled(mesh: Mesh,x:Float,y:Float,z:Float,sx:Float,sy:Float,sz:Float,colorOverride:FloatArray? = null) {
        Matrix.setIdentityM(model,0)
        Matrix.translateM(model,0,x,y,z)
        Matrix.scaleM(model,0,sx,sy,sz)
        draw(mesh,model,colorOverride)
    }

    private fun linkProgram(vs: String, fs: String): Int {
        fun shader(type:Int, src:String):Int {
            val id=GLES30.glCreateShader(type); GLES30.glShaderSource(id,src); GLES30.glCompileShader(id)
            val ok=IntArray(1); GLES30.glGetShaderiv(id,GLES30.GL_COMPILE_STATUS,ok,0)
            if(ok[0]==0) throw IllegalStateException("Shader compile failed: ${GLES30.glGetShaderInfoLog(id)}")
            return id
        }
        val p=GLES30.glCreateProgram(); GLES30.glAttachShader(p,shader(GLES30.GL_VERTEX_SHADER,vs)); GLES30.glAttachShader(p,shader(GLES30.GL_FRAGMENT_SHADER,fs)); GLES30.glLinkProgram(p)
        val ok=IntArray(1); GLES30.glGetProgramiv(p,GLES30.GL_LINK_STATUS,ok,0)
        if(ok[0]==0) throw IllegalStateException("Program link failed: ${GLES30.glGetProgramInfoLog(p)}")
        return p
    }

    private fun mesh(vertices: FloatArray, indices: ShortArray? = null): Mesh {
        val vbo=IntArray(1); GLES30.glGenBuffers(1,vbo,0); GLES30.glBindBuffer(GLES30.GL_ARRAY_BUFFER,vbo[0])
        val vb=ByteBuffer.allocateDirect(vertices.size*4).order(ByteOrder.nativeOrder()).asFloatBuffer(); vb.put(vertices).position(0)
        GLES30.glBufferData(GLES30.GL_ARRAY_BUFFER,vertices.size*4,vb,GLES30.GL_STATIC_DRAW)
        var ibo=0
        if(indices!=null) {
            val id=IntArray(1); GLES30.glGenBuffers(1,id,0); ibo=id[0]; GLES30.glBindBuffer(GLES30.GL_ELEMENT_ARRAY_BUFFER,ibo)
            val ib=ByteBuffer.allocateDirect(indices.size*2).order(ByteOrder.nativeOrder()).asShortBuffer(); ib.put(indices).position(0)
            GLES30.glBufferData(GLES30.GL_ELEMENT_ARRAY_BUFFER,indices.size*2,ib,GLES30.GL_STATIC_DRAW)
        }
        return Mesh(vbo[0],ibo,indices?.size ?: vertices.size/7)
    }

    private fun makeGround(): Mesh {
        val n=105; val extent=138f; val vertices=FloatArray((n+1)*(n+1)*7); var at=0
        for (j in 0..n) for (i in 0..n) {
            val x=-extent+2*extent*i/n; val z=-extent+2*extent*j/n
            val y=terrainHeight(x,z)
            val noise=(sin(x*0.13f+z*0.09f)*0.5f+sin(z*0.22f-x*0.17f)*0.22f)
            val radius = kotlin.math.sqrt(x*x+z*z)
            val edge=(radius/138f).coerceIn(0f,1f)
            val shore=((radius-109f)/20f).coerceIn(0f,1f)
            val sand=shore*shore*(3f-2f*shore)
            val r=(0.28f+noise*0.035f-edge*0.12f)*(1f-sand)+0.73f*sand
            val g=(0.47f+noise*0.065f-edge*0.08f)*(1f-sand)+0.66f*sand
            val b=(0.30f+noise*0.025f-edge*0.05f)*(1f-sand)+0.46f*sand
            vertices[at++]=x;vertices[at++]=y;vertices[at++]=z;vertices[at++]=r;vertices[at++]=g;vertices[at++]=b;vertices[at++]=1f
        }
        val indices=ShortArray(n*n*6); at=0
        for(j in 0 until n)for(i in 0 until n){ val a=j*(n+1)+i; val b=a+1; val c=a+n+1; val d=c+1
            indices[at++]=a.toShort();indices[at++]=c.toShort();indices[at++]=b.toShort();indices[at++]=b.toShort();indices[at++]=c.toShort();indices[at++]=d.toShort()
        }
        return mesh(vertices,indices)
    }

    private fun makeSphere(slices:Int,stacks:Int):Mesh {

    private fun terrainHeight(x:Float,z:Float):Float {
        val r=kotlin.math.sqrt(x*x+z*z);val hills=(sin(x*0.071f+1.2f)*1.0f+cos(z*0.064f)*0.83f+sin((x+z)*0.12f)*0.34f)
        val coast=((132f-r)/14f).coerceIn(0f,1f);val smooth=coast*coast*(3f-2f*coast)
        return -1.9f+(hills+1.9f)*smooth
    }

    override fun onTouchEvent(event: MotionEvent): Boolean {
        val jx=viewportWidth*0.12f;val jy=viewportHeight*0.79f;val jr=min(viewportWidth,viewportHeight)*0.13f
        when(event.actionMasked){
            MotionEvent.ACTION_DOWN,MotionEvent.ACTION_POINTER_DOWN -> {
                val i=event.actionIndex;val x=event.getX(i);val y=event.getY(i)
                if(x<viewportWidth*0.35f && y>viewportHeight*0.48f) { joystickPointer=event.getPointerId(i); updateJoystick(x,y,jx,jy,jr) }
                else { lookPointer=event.getPointerId(i); lastLookX=x;lastLookY=y }
                return true
            }
            MotionEvent.ACTION_MOVE -> {
                for(i in 0 until event.pointerCount){val id=event.getPointerId(i);val x=event.getX(i);val y=event.getY(i)
                    if(id==joystickPointer)updateJoystick(x,y,jx,jy,jr)
                    if(id==lookPointer){world.look(x-lastLookX,y-lastLookY);lastLookX=x;lastLookY=y}
                };return true
            }
            MotionEvent.ACTION_UP,MotionEvent.ACTION_POINTER_UP,MotionEvent.ACTION_CANCEL -> {
                val id=event.getPointerId(event.actionIndex)
                if(id==joystickPointer){joystickPointer=-1;world.setMove(0f,0f)}
                if(id==lookPointer){lookPointer=-1;val x=event.getX(event.actionIndex);val y=event.getY(event.actionIndex)
                    if(x>viewportWidth*0.72f && y<viewportHeight*0.43f){if(y<viewportHeight*0.2f)world.jump() else if(y<viewportHeight*0.32f)world.interact() else world.craft()}
                };return true
            }
        }
        return true
    }
    private var joystickPointer=-1;private var lookPointer=-1;private var lastLookX=0f;private var lastLookY=0f
    private fun updateJoystick(x:Float,y:Float,cx:Float,cy:Float,r:Float){val dx=x-cx;val dy=y-cy;val len=kotlin.math.sqrt(dx*dx+dy*dy).coerceAtLeast(1f);val scale=min(1f,len/r);world.setMove(dx/len*scale,-dy/len*scale)}

package ro.lingurar.hiblue.games.cozywilds

import kotlin.math.cos
import kotlin.math.sin
import kotlin.math.sqrt

/** Shared simulation state. All movement is in world units; +Z is forward at yaw 0. */
class WorldState {
    @Volatile var x = 0f
    @Volatile var z = -18f
    @Volatile var yaw = 0f
    @Volatile var pitch = -0.08f
    @Volatile var moveX = 0f
    @Volatile var moveY = 0f
    @Volatile var jumpRequested = false
    @Volatile var inventoryOpen = false
    @Volatile var dayTime = 0.23f
    @Volatile var elapsed = 0f
    @Volatile var wood = 0
    @Volatile var stone = 0
    @Volatile var berries = 0
    @Volatile var campfires = 0
    @Volatile var resourcesGathered = 0
    private val firePositions = mutableListOf<Pair<Float, Float>>()
    @Volatile var message = "A quiet morning on Fernlight Isle"
    @Volatile var messageUntil = 8f

    private var velocityY = 0f
    private var grounded = true
    private var bob = 0f
    private var lastPickup = 0f

    @Synchronized fun update(dtRaw: Float) {
        val dt = dtRaw.coerceIn(0f, 0.05f)
        elapsed += dt
        dayTime = (dayTime + dt / 420f) % 1f

        val magnitude = sqrt(moveX * moveX + moveY * moveY).coerceAtMost(1f)
        val forwardX = sin(yaw)
        val forwardZ = cos(yaw)
        val rightX = cos(yaw)
        val rightZ = -sin(yaw)
        val speed = if (magnitude > 0.08f) 5.1f else 0f
        val dx = (forwardX * moveY + rightX * moveX) * speed * dt
        val dz = (forwardZ * moveY + rightZ * moveX) * speed * dt
        val nextX = (x + dx).coerceIn(-132f, 132f)
        val nextZ = (z + dz).coerceIn(-132f, 132f)
        x = nextX
        z = nextZ
        if (magnitude > 0.08f) bob += dt * (8f + magnitude * 3f) else bob *= 0.88f

        if (jumpRequested && grounded) {
            velocityY = 5.0f
            grounded = false
        }
        jumpRequested = false
        val floorY = terrainHeight(x, z) + 1.7f
        if (grounded) {
            eyeHeight = floorY
        } else {
            velocityY -= 12.5f * dt
            eyeHeight += velocityY * dt
            if (eyeHeight <= floorY) {
                eyeHeight = floorY
                velocityY = 0f
                grounded = true
            }
        }
        if (elapsed > messageUntil) message = ""
    }

    @Volatile private var eyeHeight = 1.7f
    fun cameraY(): Float = eyeHeight + if (grounded) sin(bob) * 0.035f else 0f
    fun setMove(x: Float, y: Float) { moveX = x; moveY = y }
    fun look(dx: Float, dy: Float) {
        yaw += dx * 0.0045f
        pitch = (pitch - dy * 0.003f).coerceIn(-0.72f, 0.68f)
    }
    fun jump() { jumpRequested = true }

    @Synchronized fun interact() {
        if (elapsed - lastPickup < 0.45f) return
        lastPickup = elapsed
        // Collect a small bundle of driftwood and beach stones while exploring.
        val phase = (wood + stone + berries) % 3
        when (phase) {
            0 -> { wood += 1; toast("Picked up driftwood · Wood $wood") }
            1 -> { stone += 1; toast("Found a smooth stone · Stone $stone") }
            else -> { berries += 2; toast("Picked wild berries · Berries $berries") }
        }
        resourcesGathered++
    }

    @Synchronized fun craft() {
        if (wood >= 3 && stone >= 2) {
            wood -= 3
            stone -= 2
            campfires++
            firePositions.add(x to z)
            toast("Campfire crafted · a little warmth for tonight")
        } else {
            toast("Campfire needs 3 wood and 2 stone")
        }
    }

    @Synchronized fun campfirePositions(): List<Pair<Float, Float>> = firePositions.toList()
    fun score(): Int = resourcesGathered + campfires * 5
    @Synchronized fun toggleInventory() { inventoryOpen = !inventoryOpen }
    @Synchronized fun toast(text: String) { message = text; messageUntil = elapsed + 3.4f }

    private fun terrainHeight(x: Float, z: Float): Float {
        val r = kotlin.math.sqrt(x * x + z * z)
        val hills = (kotlin.math.sin(x * 0.071f + 1.2f) * 1.0f +
                kotlin.math.cos(z * 0.064f) * 0.83f +
                kotlin.math.sin((x + z) * 0.12f) * 0.34f)
        val coast = ((132f - r) / 14f).coerceIn(0f, 1f)
        val smooth = coast * coast * (3f - 2f * coast)
        return -1.9f + (hills + 1.9f) * smooth
    }

    fun camera(out: FloatArray) {
        out[0] = x
        out[1] = cameraY()
        out[2] = z
        out[3] = yaw
        out[4] = pitch
    }
}

        window.setFlags(WindowManager.LayoutParams.FLAG_FULLSCREEN, WindowManager.LayoutParams.FLAG_FULLSCREEN)

        val root = FrameLayout(this)
        root.addView(game, FrameLayout.LayoutParams(-1, -1))
        root.addView(hud, FrameLayout.LayoutParams(-1, -1, Gravity.TOP))
        setContentView(root)
        game.onWorldChanged = { hud.invalidate() }
    }

    override fun onBackPressed() {
        if (intent.hasExtra(EXTRA_GAME_ID)) returnResult()
        else {
            setResult(RESULT_CANCELED)
            super.onBackPressed()
        }
    }

    override fun onResume() { super.onResume(); game.onResume() }
    override fun onPause() { game.onPause(); super.onPause() }

    override fun onDestroy() {
        if (!resultReturned) setResult(RESULT_CANCELED)
        super.onDestroy()
    }

    private fun returnResult() {
        if (resultReturned) return
        resultReturned = true
        val seconds = ((System.currentTimeMillis() - startedAt) / 1000L).coerceAtMost(Int.MAX_VALUE.toLong()).toInt()
        val result = Intent()
            .putExtra("hb.game.score", game.world.score())
            .putExtra("hb.game.seconds", seconds)
        setResult(RESULT_OK, result)
        finish()
    }

    companion object {
        const val EXTRA_GAME_ID = "hb.game.id"
    }

package ro.lingurar.hiblue.nativegames

import android.content.Context
import android.content.Intent

/**
 * Un joc nativ înregistrat în host.
 *
 * runtime: "kotlin" | "godot" | "unity"
 * intentFactory: construiește Intent-ul către Activity-ul jocului (declarat în AndroidManifest de autorul jocului).
 *
 * Contract de rezultat: jocul se închide cu setResult(RESULT_OK, intent) și pune în intent
 * EXTRA_SCORE (Int) și EXTRA_SECONDS (Int). Scorul NU este de încredere: este revalidat pe server.
 */
class NativeGameEntry(
    val id: String,
    val runtime: String,
    val intentFactory: (Context) -> Intent
)

object NativeGames {
    const val EXTRA_GAME_ID = "hb.game.id"
    const val EXTRA_SCORE = "hb.game.score"
    const val EXTRA_SECONDS = "hb.game.seconds"

    private val idRe = Regex("^[a-z0-9][a-z0-9_-]{1,31}$")
    private val runtimes = setOf("kotlin", "godot", "unity")
    private val entries = LinkedHashMap<String, NativeGameEntry>()

    @Synchronized
    fun register(e: NativeGameEntry) {
        require(idRe.matches(e.id)) { "id de joc invalid: " + e.id }
        require(e.runtime in runtimes) { "runtime invalid: " + e.runtime }
        entries[e.id] = e
    }

    @Synchronized
    fun get(id: String): NativeGameEntry? = entries[id]

    @Synchronized
    fun list(): List<NativeGameEntry> = entries.values.toList()

    /**
     * Aici se înregistrează jocurile native ale aplicației.
     * Momentan NU există niciun joc nativ: lista este goală, deci lansarea unui joc nativ
     * returnează eroarea NOT_REGISTERED. Infrastructura este pregătită, nu populată.
     */
    fun registerBuiltIn() {
        register(NativeGameEntry("cozywilds", "kotlin") { context ->
            Intent(context, ro.lingurar.hiblue.games.cozywilds.MainActivity::class.java)
        })
    }
}
