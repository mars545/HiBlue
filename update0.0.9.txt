/**
 * THE JUDGMENT — Activity-ul jocului nativ (Kotlin + Filament), lansat din HiBlue prin registrul NativeGames.
 *
 * Interfața cu aplicația gazdă (vezi docs/JUDGMENT_PROTOTYPE.md):
 *   intrare : Intent din NativeGames (extra hb.game.id); opțional extra boolean "benchmark" (rotește camera 30 s și scrie perf.json)
 *   ieșire  : setResult(RESULT_CANCELED) la ieșire normală (jocul nu are încă scor); gazda emite evenimentul "nativeGame"
 *   lifecycle: onPause oprește bucla, audio și salvează; onResume le reia; onDestroy eliberează Filament + audio
 */
class JudgmentActivity : Activity(), GameEvents {
    companion object {
        private const val TAG = "Judgment"
        private const val MIN_FRAME_NANOS = 10_000_000L      // plafon ~100 FPS (pe 120 Hz randăm la 60)
        private const val BENCH_SECONDS = 30f
    }

    private lateinit var surfaceView: SurfaceView
    private lateinit var hud: HudView
    private lateinit var save: SaveStore
    private lateinit var perf: PerfMonitor
    private val input = InputState()
    private val choreographer: Choreographer by lazy { Choreographer.getInstance() }

    private var renderer: FilamentRenderer? = null
    private var audio: AudioEngine? = null
    private var world: GameWorld? = null
    private var doorAsset: FilamentAsset? = null
    private var sounds: SoundBank? = null

    private var ready = false
    private var running = false
    private var destroyedFlag = false
    private var lastFrameNanos = 0L
    private var lastRenderedNanos = 0L
    private var firstFrameLogged = false
    private var createdAt = 0L
    private var benchmark = false
    private var benchTime = 0f
    private var prevDoorState = DoorController.State.CLOSED
    private val lightFactors = FloatArray(2)
    @Volatile private var loadError: String? = null

    override fun onCreate(b: Bundle?) {
        super.onCreate(b)
        createdAt = SystemClock.elapsedRealtime()
        benchmark = intent?.getBooleanExtra("benchmark", false) ?: false
        window.addFlags(WindowManager.LayoutParams.FLAG_KEEP_SCREEN_ON)
        WindowCompat.setDecorFitsSystemWindows(window, false)
        if (android.os.Build.VERSION.SDK_INT >= 28) {
            val lp = window.attributes
            lp.layoutInDisplayCutoutMode = WindowManager.LayoutParams.LAYOUT_IN_DISPLAY_CUTOUT_MODE_SHORT_EDGES
            window.attributes = lp
        }
        save = SaveStore(this)
        perf = PerfMonitor(this)

        surfaceView = SurfaceView(this)
        hud = HudView(this, input)
        val root = FrameLayout(this)
        root.addView(surfaceView, FrameLayout.LayoutParams(-1, -1))
        root.addView(hud, FrameLayout.LayoutParams(-1, -1))
        setContentView(root)
        hideBars()

        // Filament pe firul principal, imediat (SurfaceView-ul trebuie atașat înainte de crearea suprafeței)
        val t0 = SystemClock.elapsedRealtime()
        try {
            renderer = FilamentRenderer(surfaceView)
        } catch (e: Throwable) {
            Log.e(TAG, "Inițializarea Filament a eșuat", e)
            failAndExit("Randarea 3D nu a putut porni pe acest dispozitiv")
            return
        }
        perf.mark("filament_init", SystemClock.elapsedRealtime() - t0)

        // textura / geometria / sunetul se generează în fundal; încărcarea în Filament se face apoi pe firul principal
        Thread {
            try {
                val a = LevelBuilder.build()
                val s0 = SystemClock.elapsedRealtime()
                val bank = SoundSynth.generate()
                val audioMs = SystemClock.elapsedRealtime() - s0
                runOnUiThread { onAssets(a, bank, audioMs) }
            } catch (e: Throwable) {
                Log.e(TAG, "Generarea asset-urilor a eșuat", e)
                loadError = e.toString()
                runOnUiThread { failAndExit("Generarea nivelului a eșuat") }
            }
        }.start()
    }

    private fun onAssets(a: LevelAssets, bank: SoundBank, audioMs: Long) {
        if (destroyedFlag) return
        val r = renderer ?: return
        try {
            val t0 = SystemClock.elapsedRealtime()
            r.addGlb(a.levelGlb)
            val door = r.addGlb(a.doorGlb)
            doorAsset = door
            r.createLights(LevelSpec.lights)
            perf.mark("tex_gen", a.texMs)
            perf.mark("glb_build", a.glbMs)
            perf.mark("audio_gen", audioMs)
            perf.mark("filament_load", SystemClock.elapsedRealtime() - t0)
            perf.mark("triangles", a.triangles.toLong())
        } catch (e: Throwable) {
            Log.e(TAG, "Încărcarea în Filament a eșuat", e)
            failAndExit("Nu s-a putut încărca nivelul")
            return
        }
        sounds = bank
        val w = GameWorld(this)
        save.load()?.let { w.restore(it) }
        world = w
        prevDoorState = w.door.state
        doorAsset?.let { r.setAssetTransform(it, w.door.matrix()) }
        if (running) startAudio()
        hud.loadingText = null
        ready = true
        hud.message = if (benchmark) "Benchmark 30 s…" else "THE JUDGMENT · prototip"
        Log.i(TAG, "pregătit după " + (SystemClock.elapsedRealtime() - createdAt) + " ms")
    }

    private fun startAudio() {
        if (audio == null) sounds?.let { audio = AudioEngine(this, it).also { a -> a.start() } }
        else audio?.resume()
    }

    private fun failAndExit(msg: String) {
        Log.e(TAG, msg)
        setResult(RESULT_CANCELED, Intent().putExtra(NativeGames.EXTRA_GAME_ID, "judgment"))
        finish()
    }

    private fun hideBars() {
        val c = WindowInsetsControllerCompat(window, window.decorView)
        c.systemBarsBehavior = WindowInsetsControllerCompat.BEHAVIOR_SHOW_TRANSIENT_BARS_BY_SWIPE
        c.hide(WindowInsetsCompat.Type.systemBars())
    }

    override fun onWindowFocusChanged(hasFocus: Boolean) {
        super.onWindowFocusChanged(hasFocus)
        if (hasFocus) hideBars()
    }

    // ---------------- lifecycle ----------------
    override fun onResume() {
        super.onResume()
        running = true
        lastFrameNanos = 0L
        lastRenderedNanos = 0L
        if (ready) startAudio()
        choreographer.postFrameCallback(frameCallback)
    }

    override fun onPause() {
        running = false
        choreographer.removeFrameCallback(frameCallback)
        hud.resetTouches()
        audio?.pause()
        world?.let { save.save(it.toSave()) }
        if (isFinishing) perf.writeReport(mapOf("benchmark" to benchmark.toString()))
        super.onPause()
    }

    override fun onDestroy() {
        destroyedFlag = true
        running = false
        try { audio?.release() } catch (e: Throwable) { }
        audio = null
        try { renderer?.destroy() } catch (e: Throwable) { Log.e(TAG, "destroy", e) }
        renderer = null
        super.onDestroy()
    }

    @Suppress("DEPRECATION")
    override fun onBackPressed() = exitGame()

    private fun exitGame() {
        setResult(RESULT_CANCELED, Intent().putExtra(NativeGames.EXTRA_GAME_ID, "judgment"))
        finish()
    }

    // ---------------- bucla de joc ----------------
    private val frameCallback = object : Choreographer.FrameCallback {
        override fun doFrame(frameTimeNanos: Long) {
            if (!running) return
            choreographer.postFrameCallback(this)
            if (lastFrameNanos != 0L && frameTimeNanos - lastRenderedNanos < MIN_FRAME_NANOS) return
            val interval = if (lastRenderedNanos == 0L) 16_666_667L else frameTimeNanos - lastRenderedNanos
            lastRenderedNanos = frameTimeNanos
            lastFrameNanos = frameTimeNanos
            val w0 = System.nanoTime()
            val dt = Math.min(interval, 50_000_000L) / 1e9f
            if (ready) {
                step(dt)
                renderer?.render(frameTimeNanos)
                if (!firstFrameLogged) {
                    firstFrameLogged = true
                    perf.mark("first_frame_total", SystemClock.elapsedRealtime() - createdAt)
                }
                perf.onFrame(interval, System.nanoTime() - w0)
            }
            hud.perfText = perf.overlayText
            hud.invalidate()
        }
    }

    private fun step(dt: Float) {
        val w = world ?: return
        val r = renderer ?: return
        if (input.exitPressed) { input.exitPressed = false; exitGame(); return }
        if (input.consumePerf()) hud.showPerf = !hud.showPerf
        val look = input.consumeLook()
        var yaw = look[0]
        if (benchmark) {
            yaw += 0.6f * dt
            benchTime += dt
            if (benchTime >= BENCH_SECONDS) { exitGame(); return }
        }
        if (input.consumeInteract()) w.interact()
        if (input.consumeFlash()) w.toggleFlashlight()
        val doorChanged = w.update(dt, input.strafe, input.forward, yaw, look[1])
        if (doorChanged) doorAsset?.let { r.setAssetTransform(it, w.door.matrix()) }
        val ds = w.door.state
        if (ds != prevDoorState) {
            if ((ds == DoorController.State.OPEN || ds == DoorController.State.CLOSED) &&
                (prevDoorState == DoorController.State.OPENING || prevDoorState == DoorController.State.CLOSING)) {
                audio?.playAt(Sfx.THUD, 1.5f - w.eyeX(), LevelSpec.DOOR_HINGE_Z - w.eyeZ(), w.player.yaw, 0.9f)
                w.toSave().let { save.save(it) }
            }
            prevDoorState = ds
        }
        r.setCamera(w.eyeX(), w.eyeY(), w.eyeZ(), w.camDirX(), w.camDirY(), w.camDirZ())
        lightFactors[0] = w.bulbFactor
        lightFactors[1] = w.corridorFactor
        r.updateLights(lightFactors, w.flashlightOn, w.flPosX, w.flPosY, w.flPosZ, w.flX, w.flY, w.flZ)
        hud.hint = w.hint
        hud.flashOn = w.flashlightOn
        if (hud.message != null && w.playSeconds > 5f) hud.message = null
    }

    // ---------------- GameEvents -> sunet ----------------
    override fun onDoor(opening: Boolean, x: Float, y: Float, z: Float) {
        val w = world ?: return
        audio?.playAt(Sfx.CREAK, x - w.eyeX(), z - w.eyeZ(), w.player.yaw, 1.0f)
    }

    override fun onFlashlight(on: Boolean) { audio?.play(Sfx.CLICK, 0.7f) }

    override fun onFootstep(left: Boolean) { audio?.play(if (left) Sfx.STEP1 else Sfx.STEP2, 0.35f) }
}
