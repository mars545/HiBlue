# HiBlue – update0.0.2

Ce aduce: bara de jos devine Feed · World · Community · Chat · Games; „Profile” de jos din drawer a fost scos
(rămâne cel de sus); camera Games cu coperți, rating și dezvoltator; jocul Flyby cu Top 10 în Supabase.

## Supabase (o singură dată)
SQL Editor → lipește supabase/update0.0.2.sql → Run (după hiblue.sql).
Fără el, jocul merge, dar scorul nu se trimite iar rating-ul/clasamentul nu se încarcă.

## Joc nou
Un fișier nou (sau o secțiune în games.js):  GAMES.push({id,name,tag,dev,desc,cover(k),play(host,api)});
play(host,api) întoarce {restart(),destroy()} și apelează api.over(score,secs) la final.
În Supabase: un rând în tabelul games (vezi începutul lui update0.0.2.sql).

create trigger game_ratings_touch before update on public.game_ratings for each row execute function public.game_ratings_touch();

-- media + număr voturi (toți utilizatorii)
create or replace function public.game_ratings_summary() returns table (game_id text, mean numeric, votes int)
language sql stable security definer set search_path = public as $$
  select r.game_id, round(avg(r.rating)::numeric, 1), count(*)::int
  from public.game_ratings r group by r.game_id;
$$;

-- ============ 4) Top 10 ============
create or replace function public.game_top(p_game text) returns table (pos int, user_id uuid, username text, handle text, score int)
language sql stable security definer set search_path = public as $$
  select (row_number() over (order by s.score desc, s.updated_at asc, s.user_id))::int,
         s.user_id, p.username, p.handle, s.score
  from public.game_scores s
  left join public.profiles p on p.id = s.user_id
  where s.game_id = p_game
  order by 1
  limit 10;
$$;

-- ============ 5) Trimitere scor (validează, salvează, șterge ce e sub Top 10) ============
create or replace function public.submit_score(p_game text, p_score int, p_secs int default 0) returns json
language plpgsql security definer set search_path = public as $$
declare
  uid uuid := auth.uid();
  g public.games%rowtype;
  cur int; cnt int; mn int; rk int;
  improved boolean := false;
begin
  if uid is null then raise exception 'forbidden' using errcode = '42501'; end if;
  select * into g from public.games where id = p_game;
  if not found then raise exception 'joc necunoscut' using errcode = '22023'; end if;
  if p_score < 0 or p_score > g.max_score
     or p_score > ceil(greatest(coalesce(p_secs, 0), 0) / g.pace)::int + 2 then
    raise exception 'scor invalid' using errcode = '22023';
  end if;

  perform pg_advisory_xact_lock(hashtext('score:' || p_game));

  select s.score into cur from public.game_scores s where s.game_id = p_game and s.user_id = uid;
  select count(*), min(t.score) into cnt, mn from (
    select s.score from public.game_scores s where s.game_id = p_game
    order by s.score desc, s.updated_at asc, s.user_id limit 10) t;

  if cur is null or p_score > cur then
    if cur is null and cnt >= 10 and p_score <= mn then
      return json_build_object('in_top', false, 'rank', null, 'best', null, 'min', mn, 'improved', false);
    end if;
    insert into public.game_scores (game_id, user_id, score) values (p_game, uid, p_score)
    on conflict (game_id, user_id) do update set score = excluded.score, updated_at = now();
    improved := true;
    -- ce rămâne sub Top 10 se șterge
    delete from public.game_scores d where d.game_id = p_game and d.user_id in (
      select s.user_id from public.game_scores s where s.game_id = p_game
      order by s.score desc, s.updated_at asc, s.user_id offset 10);
  end if;

  select r.pos into rk from (
    select s.user_id, (row_number() over (order by s.score desc, s.updated_at asc, s.user_id))::int as pos
    from public.game_scores s where s.game_id = p_game) r where r.user_id = uid;
  select s.score into cur from public.game_scores s where s.game_id = p_game and s.user_id = uid;
  return json_build_object('in_top', rk is not null, 'rank', rk, 'best', cur, 'min', mn, 'improved', improved);
end $$;

revoke all on function public.game_ratings_summary()          from public;
revoke all on function public.game_top(text)                  from public;
revoke all on function public.submit_score(text, int, int)    from public;
grant execute on function public.game_ratings_summary()       to authenticated;
grant execute on function public.game_top(text)               to authenticated;
grant execute on function public.submit_score(text, int, int) to authenticated;
