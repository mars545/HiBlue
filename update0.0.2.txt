  function shuf(a){var r=a.slice();for(var i=r.length-1;i>0;i--){var j=Math.floor(Math.random()*(i+1));var t=r[i];r[i]=r[j];r[j]=t}return r}
  function esc2(s){return String(s==null?"":s).replace(/[&<>"']/g,function(c){return{"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"}[c]})}

  function pickQ(list,gender){
    var pool=list;
    if(gender){
      pool=list.filter(function(it){return !it.g||it.g==='any'||it.g===gender});
      if(!pool.length)pool=list.filter(function(it){return !it.g||it.g==='any'});
    }
    if(!pool.length)return null;
    return pool[Math.floor(Math.random()*pool.length)];
  }

  function getQuestion(){
    var player=game.order[game.idx];
    var gender=player.g||'any';
    var pool=[];
    if(cfg.adult){
      cfg.adultGenres.forEach(function(g){
        var data=ADULT[g];
        if(data&&data[qType])pool.push({g:g,items:data[qType]});
      });
    }
    cfg.genres.forEach(function(g){
      var data=GENRES[g];
      if(data&&data[qType])pool.push({g:g,items:data[qType]});
    });
    if(!pool.length)return{text:"",genre:""};
    var key=qType+":"+gender;
    if(!game.usedQ[key])game.usedQ[key]=[];
    var attempts=0;
    while(attempts<12){
      var choice=pool[Math.floor(Math.random()*pool.length)];
      var item=pickQ(choice.items,gender);
      if(!item)continue;
      var text=item.t||item;
      var usedKey=choice.g+"|"+text;
      if(game.usedQ[key].indexOf(usedKey)===-1){
        game.usedQ[key].push(usedKey);
        if(game.usedQ[key].length>40)game.usedQ[key].shift();
        return{text:text,genre:choice.g};
      }
      attempts++;
    }
    var choice=pool[Math.floor(Math.random()*pool.length)];
    var item=pickQ(choice.items,gender);
    return{text:item?(item.t||item):"",genre:choice.g};
  }
