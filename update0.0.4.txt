  function loadSave() {
    try {
      const raw = localStorage.getItem(SAVE_KEY);
      if (!raw) return defaultSave();
      const saved = JSON.parse(raw);
      const base = defaultSave();
      const stats = saved.stats || {};
      return {
        ...base, ...saved,
        coins: clamp(Number(saved.coins) || 0, 0, 999999),
        stats: {
          hunger: clamp(Number(stats.hunger ?? base.stats.hunger), 0, 100),
          clean: clamp(Number(stats.clean ?? base.stats.clean), 0, 100),
          joy: clamp(Number(stats.joy ?? base.stats.joy), 0, 100),
          energy: clamp(Number(stats.energy ?? base.stats.energy), 0, 100)
        },
        owned: Array.isArray(saved.owned) ? [...new Set(saved.owned)] : base.owned,
        equipped: { ...base.equipped, ...(saved.equipped || {}) },
        decor: Array.isArray(saved.decor) ? saved.decor : [],
        themes: saved.themes || {},
        last: Number(saved.last) || Date.now()
      };
    } catch (_) {
      return defaultSave();
    }
  }

    function moveQuest(dt) {
      const q = activeQuest;
      if (!q) return;
      let ix = joy.x, iy = joy.y;
      if (keys.has("ArrowLeft") || keys.has("KeyA")) ix -= 1;
      if (keys.has("ArrowRight") || keys.has("KeyD")) ix += 1;
      if (keys.has("ArrowUp") || keys.has("KeyW")) iy -= 1;
      if (keys.has("ArrowDown") || keys.has("KeyS")) iy += 1;
      if (mapTarget) {
        const dx = mapTarget.x - q.x * W, dy = mapTarget.y - q.y * H;
        if (Math.hypot(dx, dy) < 15) mapTarget = null;
        else { ix = dx; iy = dy; }
      }
      const len = Math.hypot(ix, iy);
      if (len > .02) {
        ix /= len; iy /= len;
        const speed = q.def.vehicle ? 122 : 148;
        q.x = clamp(q.x + ix * speed * dt / W, .06, .94);

          });

        });
      }

# HiBlue – update0.0.4

Adaugă **My Blue**, un joc de tip animal de companie virtual: hrănire, baie, joacă, somn,
cinci camere, haine de cumpărat și echipat, decoruri și șapte questuri cu vedere de sus.
Progresul, monedele, ținutele și casa se salvează pe dispozitiv. Questurile încheiate trimit
scorul către clasamentul Games când există conexiune și schema Supabase este actualizată.

## Aplicare

Pune `update0.0.4.txt` în rădăcina repo-ului alături de `hiblue.txt` și update-urile anterioare,
apoi construiește APK-ul cu workflow-ul `Build APK`.

## Supabase

În SQL Editor, rulează `supabase/update0.0.4.sql` după `update0.0.3.sql` pentru rating și Top 10.
Jocul și salvarea locală funcționează și fără acest pas, însă scorurile și rating-ul nu se sincronizează.

## Controale în questuri

Folosește joystickul de pe ecran, săgețile sau tastele WASD. Pe calculator, poți și să atingi
locul în care vrei să ajungă Blue. Questul cu magazinul cere să conduci până la magazin,
să iei cele trei alimente și te întorci acasă.
