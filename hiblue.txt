# HiBlue – schelet v0.1

Autentificare / înregistrare (fără Guest) + 5 camere goale: Feed · My Blue · Games · Chat · Community.
Header: hamburger · HiBlue (cu Blue pe litera B) · [search – ascuns deocamdată] · notificări · player muzical.

## 1. Imaginile tale
Pune PNG-urile în  app/src/main/assets/img/  cu numele:
- icon.png        iconița „b”
- logo.png        logo-ul complet (corectat HiBlue)
- mascot.png      ursulețul cu fundal (ecranul de autentificare)
- mascot-cut.png  ursulețul FĂRĂ fundal (PNG transparent) – stă pe litera B din header
Dacă lipsesc, aplicația desenează variante de rezervă și funcționează oricum.

Iconița aplicației (launcher): include deja o replică vectorială a lui „b”.
Ca să o înlocuiești cu PNG-ul tău: șterge folderele mipmap-anydpi și mipmap-anydpi-v26 din res/
și pune ic_launcher.png (192×192) în res/mipmap-xxxhdpi/.

## 2. Supabase (o singură dată)
1. Dashboard → SQL Editor → lipește tot din supabase/hiblue.sql → Run.
2. Authentication → Sign In / Providers: Email = ON. „Confirm email” – ca tine (ON = primești mail de confirmare, OFF = intri direct la test).
   Anonymous sign-ins = OFF (nu mai există Guest).
3. Cheia și URL-ul sunt în app/src/main/assets/index.html, la începutul scriptului (const CFG). Acum sunt cele din Pulse;
   dacă faci un proiect Supabase nou, schimbă-le acolo.

## 3. APK (GitHub, fără instalări)
Exact ca la Pulse: repository nou → încarci TOATE fișierele → Actions → „Build APK” → Artifacts → HiBlue-apk.
Dacă folderul .github nu s-a încărcat: Add file → Create new file → nume .github/workflows/build.yml → conținutul din build-yml.txt.
Aplicația are alt applicationId (ro.lingurar.hiblue), deci se instalează lângă Pulse.

## 4. Cum umplem camerele
În index.html, secțiunile  /* ===== ROOM: ... ===== */  sunt locul fiecărei camere.
O cameră = Rooms.feed={mount(el){...}, show(){}, hide(){}}.  Căutarea din header: pune search:true la camera dorită în ROOMS.
Funcții gata pentru camere: HB.rest(...) (cereri Supabase autentificate), HB.user(), HB.toast(), HB.go(id).

package ro.lingurar.hiblue

import android.app.Activity
import android.content.Intent
import android.graphics.Color
import android.net.Uri
import android.os.Bundle
import android.webkit.*
import android.widget.Toast

class MainActivity : Activity() {
    private lateinit var wv: WebView
    private var lastBack = 0L

    override fun onCreate(b: Bundle?) {
        super.onCreate(b)
        val bg = Color.parseColor("#05050F")
        window.statusBarColor = bg
        window.navigationBarColor = bg
        wv = WebView(this)
        setContentView(wv)
        wv.setBackgroundColor(bg)
        wv.settings.apply {
            javaScriptEnabled = true
            domStorageEnabled = true
            allowFileAccess = false      // asset-urile (file:///android_asset) merg oricum
            allowContentAccess = false
            textZoom = 100               // interfața nu se strică la fontul mare din telefon
            setSupportZoom(false)
            mediaPlaybackRequiresUserGesture = false
        }
        wv.overScrollMode = android.view.View.OVER_SCROLL_NEVER
        wv.webViewClient = object : WebViewClient() {
            override fun shouldOverrideUrlLoading(v: WebView?, r: WebResourceRequest?): Boolean {
                val u = r?.url?.toString() ?: return false
                if (u.startsWith("file:///android_asset") || u.startsWith("about:") || u.startsWith("data:") || u.startsWith("blob:")) return false
                try { startActivity(Intent(Intent.ACTION_VIEW, Uri.parse(u))) } catch (e: Exception) {}
                return true
            }
        }
        wv.webChromeClient = object : WebChromeClient() {
            override fun onConsoleMessage(m: ConsoleMessage?): Boolean {
                m?.let { android.util.Log.d("HiBlueJS", "[${it.messageLevel()}] ${it.message()} @${it.sourceId()}:${it.lineNumber()}") }
                return true
            }
        }
        if (b != null) wv.restoreState(b) else wv.loadUrl("file:///android_asset/index.html")
    }

    override fun onSaveInstanceState(o: Bundle) { super.onSaveInstanceState(o); wv.saveState(o) }

    // Back: întâi lasă interfața să decidă (închide drawer-ul / revine la Feed), apoi dublu-back pentru ieșire
    @Suppress("DEPRECATION")
    override fun onBackPressed() {
        wv.evaluateJavascript("(window.HB&&HB.back)?HB.back():false") { r ->
            if (r == "true") return@evaluateJavascript
            val now = System.currentTimeMillis()
            if (now - lastBack < 2000) finish()
            else { lastBack = now; Toast.makeText(this, "Apasă încă o dată pentru a ieși", Toast.LENGTH_SHORT).show() }
        }
    }

    override fun onDestroy() { try { wv.destroy() } catch (e: Exception) {}; super.onDestroy() }
}

-- ============ 1) PROFILE ============
create table if not exists public.profiles (
  id         uuid primary key references auth.users(id) on delete cascade,
  username   text,                       -- numele afișat
  handle     text,                       -- @utilizator (unic)
  bio        text,
  avatar     text,                       -- emoji sau cale/URL (în update-urile viitoare)
  created_at timestamptz not null default now()
);
alter table public.profiles add column if not exists updated_at timestamptz not null default now();

alter table public.profiles drop constraint if exists profiles_handle_format;
alter table public.profiles add  constraint profiles_handle_format
  check (handle is null or handle ~ '^[a-z0-9_.]{3,20}$');
create unique index if not exists profiles_handle_lower_key on public.profiles (lower(handle));

-- limite de lungime (le înlocuiesc pe cele strânse din Pulse: avatar putea avea doar 8 caractere)
alter table public.profiles drop constraint if exists profiles_len;
alter table public.profiles drop constraint if exists profiles_len_v2;
alter table public.profiles add  constraint profiles_len_v2
  check (char_length(coalesce(username,'')) <= 32
     and char_length(coalesce(bio,''))      <= 160
     and char_length(coalesce(avatar,''))   <= 500);

-- ============ 2) RLS ============
alter table public.profiles enable row level security;

create policy profiles_select on public.profiles for select to authenticated using (true);

drop policy if exists profiles_insert on public.profiles;
create policy profiles_insert on public.profiles for insert to authenticated with check (auth.uid() = id);

drop policy if exists profiles_update on public.profiles;

  using (auth.uid() = id) with check (auth.uid() = id);

begin
  new.updated_at = now();
  return new;
end $$;

-- ============ 4) Profil creat automat la înregistrare ============
-- Aplicația trimite username + handle în metadatele înregistrării; trigger-ul le preia.
-- Merge și când contul cere confirmare pe email (încă nu există sesiune, deci clientul nu poate scrie singur).
create or replace function public.handle_new_user() returns trigger
language plpgsql security definer set search_path = public as $$
declare
  h text := lower(nullif(trim(new.raw_user_meta_data->>'handle'), ''));
  n text := nullif(trim(new.raw_user_meta_data->>'username'), '');
begin
  if coalesce(new.is_anonymous, false) then return new; end if;
  if n is not null then n := left(n, 32); end if;
  if h is not null and h !~ '^[a-z0-9_.]{3,20}$' then h := null; end if;
  if h is not null and exists (select 1 from public.profiles where lower(handle) = h) then h := null; end if;
  begin
    insert into public.profiles(id, username, handle) values (new.id, n, h) on conflict (id) do nothing;
  exception when unique_violation then
    insert into public.profiles(id, username, handle) values (new.id, n, null) on conflict (id) do nothing;
  end;
  return new;
end $$;

drop trigger if exists on_auth_user_created on auth.users;
create trigger on_auth_user_created after insert on auth.users
  for each row execute function public.handle_new_user();

-- ============ 5) Verificare @utilizator liber (apelată de ecranul de înregistrare) ============
create or replace function public.handle_available(h text) returns boolean
language sql stable security definer set search_path = public as $$
  select h ~ '^[a-z0-9_.]{3,20}$'
     and not exists (select 1 from public.profiles where lower(handle) = lower(h));
$$;
revoke all on function public.handle_available(text) from public;
grant execute on function public.handle_available(text) to anon, authenticated;
