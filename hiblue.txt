/* Tax brackets (Romania-inspired, simplified) */
const TAX_BRACKETS=[
  {upTo:3000, rate:0.10, deduct:0},
  {upTo:6000, rate:0.20, deduct:300},
  {upTo:12000,rate:0.30, deduct:900},
  {upTo:Infinity,rate:0.40, deduct:2100},
];
const CAS_RATE=0.35; /* social contribution approx */

function netFromGross(gross){
  /* CAS on gross, then income tax on remainder */
  const cas=gross*CAS_RATE;
  const taxable=gross-cas;
  const bracket=TAX_BRACKETS.find(b=>gross<=b.upTo);
  const tax=taxable*bracket.rate - bracket.deduct;
  return Math.max(0, gross-cas-tax);
}

function buildWorld(){
  const world=$('#world');
  world.innerHTML='';

  /* Regions */
  REGIONS.forEach(r=>{
    const el=document.createElement('div');
    el.className='region '+r.cls;
    el.style.left=getRegionStyle(r.id).left;
    el.style.top=getRegionStyle(r.id).top;
    el.style.width=getRegionStyle(r.id).width;
    el.style.height=getRegionStyle(r.id).height;
    el.innerHTML=`<div class="region-label">${r.name}<small>${r.sub}</small></div>`;
    world.appendChild(el);
  });

  /* Lake */
  const lake=document.createElement('div');
  lake.className='lake';
  world.appendChild(lake);

  /* Roads */
  ROADS.forEach(rd=>{
    const r=document.createElement('div');
    r.className='road '+rd.cls;
    r.style.left=rd.x+'px';
    r.style.top=rd.y+'px';
    if(rd.cls==='h') r.style.width=rd.w+'px';
    else r.style.height=rd.h+'px';
    world.appendChild(r);
  });

  /* Trees scattered */
  for(let i=0;i<60;i++){
    const t=document.createElement('div');
    t.className='tree';
    /* place in grass-only areas (outside regions roughly) */
    let tx,ty,ok=false,tries=0;
    while(!ok && tries<30){
      tx=rndi(50,2350); ty=rndi(50,1750);
      ok = !isInsideRegion(tx,ty) || inRegionId(tx,ty)==='silverlake';
      tries++;
    }
    if(ok){
      t.style.left=tx+'px';t.style.top=ty+'px';
      t.style.transform=`scale(${rnd(.7,1.3)})`;
      world.appendChild(t);
    }
  }

  /* Buildings */
  WORLD_BUILDINGS.forEach(b=>{
    const el=document.createElement('div');
    el.className=`building b-${b.size==='b-sm'?'sm':b.size==='b-md'?'md':b.size==='b-lg'?'lg':'xl'} t-${b.type}`;
    el.dataset.id=b.id;
    el.style.left=b.x+'px';
    el.style.top=b.y+'px';
    el.innerHTML=`<div class="roof"></div><div class="body"></div><div class="ico">${b.icon}</div>`;
    el.title=b.name;
    el.addEventListener('click',()=>selectBuilding(b.id));
    world.appendChild(el);
  });

  SPECIAL_PLOTS.forEach(p=>{
    const pos=getSpecialPos(p.id);

    el.className='building b-md t-land special';

    el.title=p.name+' — '+p.area+' mp · '+p.zone;
    el.addEventListener('click',()=>openSpecialPlot(p.id));
    world.appendChild(el);
  });
}

/* Helper: region position lookup (sync with CSS classes above) */
function getRegionStyle(id){
  const map={
    bluehaven:  {left:'520px',  top:'200px',  width:'1300px', height:'980px'},
    westport:   {left:'1880px', top:'760px',  width:'480px',  height:'780px'},
    pinecrest:  {left:'80px',   top:'1080px', width:'560px',  height:'560px'},
    littlebriar:{left:'660px',  top:'1240px', width:'560px',  height:'460px'},
    greenfield: {left:'1280px', top:'1180px', width:'540px',  height:'520px'},
    silverlake: {left:'1880px', top:'60px',   width:'480px',  height:'620px'},
  };
  return map[id]||{left:'0',top:'0',width:'0',height:'0'};
}
function getRegionRect(id){
  const s=getRegionStyle(id);
  return {x:parseInt(s.left),y:parseInt(s.top),w:parseInt(s.width),h:parseInt(s.height)};
}
function isInsideRegion(x,y){
  return REGIONS.some(r=>{
    const rc=getRegionRect(r.id);
    return x>=rc.x && x<=rc.x+rc.w && y>=rc.y && y<=rc.y+rc.h;
  });
}
function inRegionId(x,y){
  for(const r of REGIONS){
    const rc=getRegionRect(r.id);
    if(x>=rc.x && x<=rc.x+rc.w && y>=rc.y && y<=rc.y+rc.h) return r.id;
  }
  return null;
}

function getSpecialPos(id){
  /* Distribute special plots visibly inside their region */
  const map={
    sp1:{x:1500,y:680}, sp2:{x:600,y:1000}, sp3:{x:1960,y:1300},
    sp4:{x:480, y:1180}, sp5:{x:1100,y:1380}, sp6:{x:1340,y:1500},
    sp7:{x:1960,y:60},  sp8:{x:2200,y:80},
  };
  return map[id];
}

let worthHistory=[];
function drawWorthChart(currentWorth){
  worthHistory.push(currentWorth);
  if(worthHistory.length>14) worthHistory.shift();
  const el=$('#worthChart');
  if(!el) return;
  const w=el.clientWidth-16, h=el.clientHeight-16;
  const max=Math.max(...worthHistory,1), min=Math.min(...worthHistory,0);
  const range=Math.max(1,max-min);
  const pts=worthHistory.map((v,i)=>`${(i/(worthHistory.length-1||1))*w},${h-((v-min)/range)*h}`).join(' ');
  el.innerHTML=`<svg width="100%" height="100%" viewBox="0 0 ${w} ${h}" preserveAspectRatio="none">
    <polyline fill="none" stroke="#1f7ae0" stroke-width="2" points="${pts}"/>
    <polyline fill="rgba(31,122,224,.15)" stroke="none" points="0,${h} ${pts} ${w},${h}"/>
  </svg>`;
}

function selectBuilding(id){

function buyProperty(bid){
  const b=WORLD_BUILDINGS.find(x=>x.id===bid);
  if(!b) return;
  const h=HOUSING[b.housingIdx];
  if(STATE.player.cash<h.price){toast('Fonduri insuficiente!','bad');return;}
  STATE.player.cash-=h.price;
  STATE.economy.totalSpent+=h.price;
  const prop={
    id:bid, name:b.name, region:b.region, type:b.type,
    housingIdx:b.housingIdx, value:h.price, condition:100,
    purchasePrice:h.price, purchaseDay:STATE.world.day,
  };
  STATE.player.ownedProperties.push(prop);
  markOwned(bid);

function applyLoan(i){
  const types=[
    {type:'Credit personal',      max:20000,  rate:0.045},
    {type:'Credit ipotecar',      max:500000, rate:0.018},
    {type:'Credit pentru afaceri',max:500000, rate:0.025},
    {type:'Credit pentru utilaje',max:100000, rate:0.030},
    {type:'Credit pentru extindere',max:200000,rate:0.028},
  ];
  const c=types[i];
  const creditFactor=clamp(STATE.player.creditScore/700,0.3,1.5);
  const maxAllowed=Math.min(c.max,Math.max(2000,c.max*creditFactor));
  const amt=prompt(`Solicită ${c.type}. Maxim disponibil: ${fmt(maxAllowed)}. Sumă dorită:`,String(Math.round(maxAllowed/2)));
  if(!amt) return;
  const v=Math.max(0,parseFloat(amt)||0);
  if(v>maxAllowed){toast('Sumă peste limita aprobată!','bad');return;}
  const installments=36;
  const total=v*(1+c.rate*installments);
  const monthlyPayment=total/installments;
  STATE.player.cash+=v;
  STATE.player.bank+=0;
  STATE.player.debt+=total;
  STATE.player.activeLoans.push({
    id:uid('loan'),type:c.type,principal:v,remaining:total,rate:c.rate,
    monthlyPayment,installmentsLeft:installments,
  });
  STATE.player.creditScore=clamp(STATE.player.creditScore-10,300,850);
  toast(`Împrumut aprobat: ${fmt(v)}. Rata: ${fmt(monthlyPayment)}/lună.`,'good');
  news(`Credit nou: ${c.type} · ${fmt(v)}`,'warn');
  refreshHUD(); openPanel('bank');
}
function payLoan(lid){
  const l=STATE.player.activeLoans.find(x=>x.id===lid);
  if(!l) return;
  if(STATE.player.cash<l.monthlyPayment){toast('Fonduri insuficiente!','bad');return;}
  STATE.player.cash-=l.monthlyPayment;
  l.remaining-=l.monthlyPayment;
  l.installmentsLeft--;
  STATE.player.debt-=l.monthlyPayment;
  STATE.economy.totalSpent+=l.monthlyPayment;
  STATE.player.creditScore=clamp(STATE.player.creditScore+3,300,850);
  if(l.remaining<=0||l.installmentsLeft<=0){
    STATE.player.activeLoans=STATE.player.activeLoans.filter(x=>x.id!==lid);
    toast(`Credit rambursat complet!`,'good');
    news(`Credit închis: ${l.type}`,'good');
  } else {
    toast(`Rată plătită. Restant: ${fmt(l.remaining)}`,'good');
  }
  refreshHUD(); openPanel('bank');
}

/* ============================================================
   TIME & ECONOMY SIMULATION
   ============================================================ */
let lastTick=0;
const TICK_MS_NORMAL=2000; /* 1 game hour = 2 real seconds at normal speed */
const TICK_MS_FAST=600;
const HOURS_PER_DAY=24;

function gameLoop(now){
  if(!lastTick) lastTick=now;
  const speed=STATE.world.speed;
  if(speed===0){requestAnimationFrame(gameLoop);return;}
  const interval=speed===2?TICK_MS_FAST:TICK_MS_NORMAL;
  if(now-lastTick>=interval){
    lastTick=now;
    advanceHour();
  }
  requestAnimationFrame(gameLoop);
}

function advanceHour(){
  STATE.world.hour++;
  if(STATE.world.hour>=HOURS_PER_DAY){
    STATE.world.hour=0;
    advanceDay();
  }
  updateStageVisual();
  refreshHUD();
}

function advanceDay(){
  STATE.world.day++;
  /* Season change every 30 days */
  if(STATE.world.day%30===1 && STATE.world.day>1){
    STATE.world.seasonIdx=(STATE.world.seasonIdx+1)%4;
    STATE.world.season=['Primăvară','Vară','Toamnă','Iarnă'][STATE.world.seasonIdx];
    news(`Anotimp nou: ${STATE.world.season}`,'warn');
  }
  /* Random weather */
  if(Math.random()<0.25){
    STATE.world.weather=pick(['Senin','Înnorat','Ploios','Înnorat','Senin']);
  }

  /* Energy decay if no home */
  if(!STATE.player.rentHome && !STATE.player.ownedProperties.length){
    STATE.player.energy=clamp(STATE.player.energy-15,0,100);
    if(STATE.player.energy<=0){
      STATE.player.health=clamp(STATE.player.health-10,0,100);
      news('Epuizare: ai nevoie de o locuință!','bad');
    }
  } else {
    STATE.player.energy=clamp(STATE.player.energy+20,0,100);
  }

  /* Job income & XP */
  let dailyIncome=0, dailyExpense=0;
  if(STATE.player.job){
    const netDaily=netFromGross(STATE.player.job.salary)/30;
    dailyIncome+=netDaily;
    /* Job XP for working days (Mon-Sat = 6 days/week) */
    const dow=STATE.world.day%7;
    if(dow!==0){ /* Sunday = 0 = rest */
      STATE.player.jobXp+=10;
      STATE.player.skills[STATE.player.job.domain]=(STATE.player.skills[STATE.player.job.domain]||0)+8;
      if(STATE.player.jobXp>=STATE.player.jobXpNeeded){
        STATE.player.jobXp=0;
        STATE.player.jobLevel++;
        STATE.player.jobXpNeeded=Math.round(STATE.player.jobXpNeeded*1.4);
        STATE.player.job.salary*=(1+0.08);
        news(`Promovare! Nivel ${STATE.player.jobLevel} · Salariu ${fmt(STATE.player.job.salary)}`,'good');
      }
    }
  }

  /* Rental income from rented-out properties */
  STATE.player.rentedOut.forEach(r=>{
    dailyIncome+=r.rent/30;
  });

  /* Business income */
  STATE.player.ownedBusinesses.forEach(b=>{
    /* Income: random within range, modified by stock and staff ratio */
    const staffRatio=Math.min(1,b.staff.length/b.staffNeeded);
    const stockMul=b.stock>30?1:0.4;
    const markupMul=1+ (b.markup-1.3)*(-0.4); /* higher markup reduces volume */
    let income=rnd(b.dailyIncomeRange[0],b.dailyIncomeRange[1])*staffRatio*stockMul*markupMul;
    income=Math.max(0,income);
    /* Daily expense: salaries/30 + utilities */
    const salaries=b.staff.reduce((s,x)=>s+x.salary,0)/30;
    const utilities=b.value*0.0005; /* 0.05% of value per day */
    const expense=salaries+utilities;
    /* Reduce stock */
    b.stock=Math.max(0,b.stock-Math.round(income/20));
    b.lastIncome=income;
    b.lastExpense=expense;
    b.lastProfit=income-expense;
    dailyIncome+=income;
    dailyExpense+=expense;
  });

  /* Housing expenses */
  STATE.player.ownedProperties.forEach(p=>{
    const h=HOUSING[p.housingIdx];
    const maint=h.rent*0.25/30;
    dailyExpense+=maint;
    /* Property decay */
    p.condition=clamp(p.condition-rnd(0.1,0.4),0,100);
  });

  /* Rent for rented home */
  if(STATE.player.rentHome && STATE.player.rentHome.kind==='rent'){
    dailyExpense+=STATE.player.rentHome.rent/30;
  }

  /* Loan payments (monthly, but daily fraction) */
  STATE.player.activeLoans.forEach(l=>{
    dailyExpense+=l.monthlyPayment/30;
    l.remaining-=l.monthlyPayment/30;
  });
  /* If can't pay loans, reduce credit score */
  if(STATE.player.activeLoans.length>0 && STATE.player.cash+STATE.player.bank<0){
    STATE.player.creditScore=clamp(STATE.player.creditScore-2,300,850);
  }

  /* Savings interest */
  if(STATE.player.savings>0){
    const interest=STATE.player.savings*0.03/30;
    STATE.player.savings+=interest;
    dailyIncome+=interest;
  }

  /* Inflation slow creep */
  STATE.world.inflation=clamp(STATE.world.inflation+0.0002,0.95,1.20);

  /* Market trends: 5% chance each product shifts */
  PRODUCTS.forEach(p=>{
    if(Math.random()<0.10){
      const cur=STATE.world.marketTrend[p.name]||1;
      const next=clamp(cur*(1+rnd(-0.08,0.08)),0.5,2.0);
      STATE.world.marketTrend[p.name]=next;
    }
  });

  /* Apply totals */
  STATE.player.cash+=dailyIncome-dailyExpense;
  if(STATE.player.cash<0){
    /* Auto-cover from bank if possible */
    if(STATE.player.bank>Math.abs(STATE.player.cash)){
      STATE.player.bank+=STATE.player.cash;
      STATE.player.cash=0;
    } else {
      /* Take from savings */
      const need= Math.abs(STATE.player.cash);
      const fromBank=Math.min(STATE.player.bank,need);
      STATE.player.bank-=fromBank;
      let remaining=need-fromBank;
      const fromSavings=Math.min(STATE.player.savings,remaining);
      STATE.player.savings-=fromSavings;
      remaining-=fromSavings;
      STATE.player.cash=-remaining;
      if(remaining>0){
        news(`Deficit zilnic ${fmt(remaining)} — risc credit!`,'bad');
        STATE.player.creditScore=clamp(STATE.player.creditScore-3,300,850);
      }
    }
  }
  STATE.economy.totalEarned+=dailyIncome;
  STATE.economy.totalSpent+=dailyExpense;
  STATE.economy.lastDailyIncome=dailyIncome;
  STATE.economy.lastDailyExpense=dailyExpense;
  STATE.economy.lastDailyProfit=dailyIncome-dailyExpense;

  /* Random events */
  if(Math.random()<0.04){
    randomEvent();
  }

  refreshHUD();
}

/* ============================================================
   PAN & ZOOM on world
   ============================================================ */
let pan={x:0,y:0,scale:1,dragging:false,sx:0,sy:0};
function setupPan(){
  const stage=$('#stage');
  const world=$('#world');
  stage.addEventListener('mousedown',startPan);
  stage.addEventListener('touchstart',startPan,{passive:false});
  window.addEventListener('mousemove',movePan);
  window.addEventListener('touchmove',movePan,{passive:false});
  window.addEventListener('mouseup',endPan);
  window.addEventListener('touchend',endPan);
  /* Wheel zoom (desktop) */
  stage.addEventListener('wheel',e=>{
    e.preventDefault();
    pan.scale=clamp(pan.scale*(e.deltaY<0?1.08:0.92),0.5,2.0);
    applyTransform();
  },{passive:false});
}
function startPan(e){
  if(e.target.closest('.building')) return; /* let building clicks pass */
  pan.dragging=true;
  const p=pt(e);
  pan.sx=p.x-pan.x;
  pan.sy=p.y-pan.y;
  document.body.style.cursor='grabbing';
}
function movePan(e){
  if(!pan.dragging) return;
  e.preventDefault?.();
  const p=pt(e);
  pan.x=p.x-pan.sx;
  pan.y=p.y-pan.sy;
  applyTransform();
}
function endPan(){
  pan.dragging=false;
  document.body.style.cursor='';
}
function pt(e){
  if(e.touches&&e.touches[0]) return{x:e.touches[0].clientX,y:e.touches[0].clientY};
  return{x:e.clientX,y:e.clientY};
}
function applyTransform(){
  const w=$('#world');
  const isPortrait=window.matchMedia('(max-aspect-ratio:1/1)').matches;
  const tilt=isPortrait?30:42;
  const baseScale=isPortrait?.42:.55;
  const s=baseScale*pan.scale;
  w.style.transform=`translate(calc(-50% + ${pan.x}px), calc(-50% + ${pan.y}px)) perspective(2200px) rotateX(${tilt}deg) scale(${s})`;
}
function panTo(wx,wy){
  /* Center world coordinate wx,wy on screen.
     Effective on-screen scale = baseScale * pan.scale (default 0.55 * 1 = 0.55). */
  const cx=window.innerWidth/2, cy=window.innerHeight/2;
  const effScale=0.55*pan.scale;
  pan.x=cx-(wx*effScale);
  pan.y=cy-(wy*effScale);
  applyTransform();
}

    const s=localStorage.getItem(SAVE_KEY);

}

  location.reload();
}
