# HiBlue · versiunea 0.0.12

HiBlue este o aplicație Android hibridă: Android/Kotlin ca host, interfață HTML/CSS/JavaScript în WebView, GamePlatform pentru jocuri și Supabase pentru serviciile online.

## Jocuri active
- **Flyby** — arcade, runtime web.
- **Flip Nova** — arcade, runtime web.

În această versiune au fost eliminate complet cele trei prototipuri abandonate, împreună cu resursele și înregistrările specifice acestora. Infrastructura generală GamePlatform și adaptorul generic pentru jocuri native au fost păstrate.

## Build APK
Workflow-ul GitHub Actions **Build APK** reconstruiește proiectul pornind de la `hiblue.txt` și aplică update-urile în ordine. Pentru versiunea 0.0.12, build-ul produce APK-ul și noul artefact `hiblue-txt-0.0.12`.

## Arhitectură și extensii
Documentația tehnică se află în `docs/`. Pentru un joc web nou, urmează `docs/ADDING_WEB_GAME.md` și `docs/GAME_SDK.md`. Pentru integrarea unui joc Android nativ, vezi `docs/NATIVE_GAMES.md`.

## Notă privind datele externe
Acest update elimină codul jocurilor din proiectul Android. Nu șterge automat rânduri, scoruri sau ratinguri din proiectul Supabase deja configurat și nu șterge datele istorice din Git.

package ro.lingurar.hiblue

import android.Manifest
import android.app.Activity
import android.content.Intent
import android.content.pm.ActivityInfo
import android.content.pm.ApplicationInfo
import android.content.pm.PackageManager
import android.graphics.Color
import android.net.Uri
import android.os.Build
import android.os.Bundle
import android.view.ViewGroup
import android.view.WindowManager
import android.webkit.*
import android.widget.FrameLayout
import android.widget.Toast
import androidx.core.view.ViewCompat
import androidx.core.view.WindowCompat
import androidx.core.view.WindowInsetsCompat
import androidx.core.view.WindowInsetsControllerCompat
import org.json.JSONObject
import ro.lingurar.hiblue.bridge.HostActions
import ro.lingurar.hiblue.bridge.HostBridge
import ro.lingurar.hiblue.host.DeviceServices
import ro.lingurar.hiblue.nativegames.NativeGames

/**
 * Android Host: ciclul de viață, WebView, inset-uri, Back, permisiuni și puntea către frontend.
 * Interfața (UI) rămâne în aplicația web din assets; aici NU se construiește UI.
 *
 * Comunicare:
 *   JS -> Kotlin : window.HBNative.invoke(json)   (vezi bridge/HostBridge.kt)
 *   Kotlin -> JS : NativeBridge._emit(tip, date)  (lifecycle, permission, nativeGame)
 */
class MainActivity : Activity() {
    private companion object {
        const val ASSET_PREFIX = "file:///android_asset/"
        const val START_URL = "file:///android_asset/index.html"
        const val REQ_NOTIF = 7
        const val REQ_NATIVE_GAME = 9
    }

    private lateinit var wv: WebView
    private lateinit var device: DeviceServices
    private var lastBack = 0L
    private var insTop = 0
    private var insBottom = 0
    @Volatile private var pendingNativeId: String? = null

    /** Setat doar din firul UI (onPageStarted); citit din firul JavaBridge. */
    @Volatile private var pageTrusted = false

    private val actions = object : HostActions {
        override fun trustedPage(): Boolean = pageTrusted

        override fun vibrate(kind: Int) {
            runOnUiThread { device.vibrate(kind) }
        }

        override fun notificationsGranted(): Boolean = device.notificationsGranted()

        override fun requestNotifications() {
            runOnUiThread {
                if (Build.VERSION.SDK_INT >= 33 && !device.notificationsGranted()) {
                    requestPermissions(arrayOf(Manifest.permission.POST_NOTIFICATIONS), REQ_NOTIF)
                }
            }
        }

        override fun notify(title: String, body: String): Boolean = device.notify(title, body)

        override fun setOrientation(mode: String): Boolean {
            val v = when (mode) {
                "portrait" -> ActivityInfo.SCREEN_ORIENTATION_PORTRAIT
                "landscape" -> ActivityInfo.SCREEN_ORIENTATION_SENSOR_LANDSCAPE
                "auto" -> ActivityInfo.SCREEN_ORIENTATION_FULL_SENSOR
                else -> return false
            }
            runOnUiThread { requestedOrientation = v }
            return true
        }

        override fun nativeGames(): List<Pair<String, String>> =
            NativeGames.list().map { Pair(it.id, it.runtime) }

        override fun launchNativeGame(id: String): String? {
            val entry = NativeGames.get(id) ?: return "NOT_REGISTERED"
            if (pendingNativeId != null) return "BUSY"
            pendingNativeId = id
            runOnUiThread {
                try {
                    val i = entry.intentFactory(this@MainActivity)
                    i.putExtra(NativeGames.EXTRA_GAME_ID, id)
                    @Suppress("DEPRECATION")
                    startActivityForResult(i, REQ_NATIVE_GAME)
                } catch (e: Exception) {
                    pendingNativeId = null
                    emit("nativeGame", JSONObject().put("id", id).put("status", "error"))
                }
            }
            return null
        }
    }

    override fun onCreate(b: Bundle?) {
        super.onCreate(b)
        NativeGames.registerBuiltIn()
        device = DeviceServices(this)

        val bg = Color.parseColor("#05050F")
        window.statusBarColor = bg
        window.navigationBarColor = bg
        if (Build.VERSION.SDK_INT >= 28) {
            val lp = window.attributes
            lp.layoutInDisplayCutoutMode = WindowManager.LayoutParams.LAYOUT_IN_DISPLAY_CUTOUT_MODE_SHORT_EDGES
            window.attributes = lp
        }
        WindowCompat.setDecorFitsSystemWindows(window, false)

        wv = WebView(this)
        val root = FrameLayout(this)
        root.setBackgroundColor(bg)
        root.addView(wv, FrameLayout.LayoutParams(ViewGroup.LayoutParams.MATCH_PARENT, ViewGroup.LayoutParams.MATCH_PARENT))
        setContentView(root)
        wv.setBackgroundColor(bg)

        // Tastatura împinge conținutul în sus; marginile (notch / bare) ajung în CSS ca --sat / --sab
        ViewCompat.setOnApplyWindowInsetsListener(root) { v, ins ->
            val ime = ins.getInsets(WindowInsetsCompat.Type.ime())
            val bars = ins.getInsets(WindowInsetsCompat.Type.systemBars() or WindowInsetsCompat.Type.displayCutout())
            v.setPadding(0, 0, 0, ime.bottom)
            insTop = bars.top
            insBottom = if (ime.bottom > 0) 0 else bars.bottom
            pushInsets()
            WindowInsetsCompat.CONSUMED
        }

        configureWebView()
        wv.addJavascriptInterface(HostBridge(actions, appVersion()), HostBridge.NAME)
        wv.webViewClient = object : WebViewClient() {
            override fun onPageStarted(v: WebView?, url: String?, favicon: android.graphics.Bitmap?) {
                pageTrusted = url != null && url.startsWith(ASSET_PREFIX)
            }

            override fun onPageFinished(v: WebView?, url: String?) { pushInsets() }

            override fun shouldOverrideUrlLoading(v: WebView?, r: WebResourceRequest?): Boolean {
                val u = r?.url?.toString() ?: return true
                if (u.startsWith(ASSET_PREFIX) || u == "about:blank") return false
                // Conținutul extern nu se încarcă niciodată în WebView-ul cu punte: se deschide în browser.
                if (u.startsWith("https://") || u.startsWith("http://") || u.startsWith("mailto:") || u.startsWith("tel:")) {
                    try { startActivity(Intent(Intent.ACTION_VIEW, Uri.parse(u))) } catch (e: Exception) {}
                }
                return true
            }

            override fun onRenderProcessGone(view: WebView?, detail: RenderProcessGoneDetail?): Boolean {
                // Procesul de randare a murit (memorie / crash): refacem activitatea în loc să cădem aplicația.
                pageTrusted = false
                Toast.makeText(this@MainActivity, "Aplicația a fost repornită", Toast.LENGTH_SHORT).show()
                recreate()
                return true
            }
        }
        wv.webChromeClient = object : WebChromeClient() {
            override fun onConsoleMessage(m: ConsoleMessage?): Boolean {
                m?.let { android.util.Log.d("HiBlueJS", "[${it.messageLevel()}] ${it.message()} @${it.sourceId()}:${it.lineNumber()}") }
                return true
            }
        }
        if (b != null) wv.restoreState(b) else wv.loadUrl(START_URL)
        hideBars()
    }

    private fun configureWebView() {
        if ((applicationInfo.flags and ApplicationInfo.FLAG_DEBUGGABLE) != 0) {
            WebView.setWebContentsDebuggingEnabled(true)
        }
        wv.settings.apply {
            javaScriptEnabled = true
            domStorageEnabled = true
            allowFileAccess = false
            allowContentAccess = false
            allowFileAccessFromFileURLs = false
            allowUniversalAccessFromFileURLs = false
            javaScriptCanOpenWindowsAutomatically = false
            setSupportMultipleWindows(false)
            setGeolocationEnabled(false)
            mixedContentMode = WebSettings.MIXED_CONTENT_NEVER_ALLOW
            textZoom = 100
            setSupportZoom(false)
            mediaPlaybackRequiresUserGesture = false
        }
        wv.overScrollMode = android.view.View.OVER_SCROLL_NEVER
    }

    private fun appVersion(): String = try {
        packageManager.getPackageInfo(packageName, 0).versionName ?: "?"
    } catch (e: Exception) {
        "?"
    }

    /** Kotlin -> JS. Apelat doar din firul UI. Datele sunt JSON controlat de host (numere / id-uri). */
    private fun emit(type: String, data: JSONObject) {
        val js = "window.NativeBridge&&NativeBridge._emit(" + JSONObject.quote(type) + "," + data.toString() + ")"
        wv.evaluateJavascript(js, null)
    }

    private fun pushInsets() {
        val d = resources.displayMetrics.density
        wv.evaluateJavascript("var s=document.documentElement.style;s.setProperty('--sat','${insTop / d}px');s.setProperty('--sab','${insBottom / d}px')", null)
    }

    /** Full screen: bare de sistem ascunse (apar la swipe de la margine) */
    private fun hideBars() {
        val c = WindowInsetsControllerCompat(window, window.decorView)
        c.systemBarsBehavior = WindowInsetsControllerCompat.BEHAVIOR_SHOW_TRANSIENT_BARS_BY_SWIPE
        c.hide(WindowInsetsCompat.Type.systemBars())
    }

    override fun onWindowFocusChanged(hasFocus: Boolean) {
        super.onWindowFocusChanged(hasFocus)
        if (hasFocus) hideBars()
    }

    override fun onPause() {
        // Întâi anunțăm frontend-ul (jocurile se opresc), apoi oprim WebView-ul.
        if (pageTrusted) emit("lifecycle", JSONObject().put("state", "pause"))
        wv.onPause()
        super.onPause()
    }

    override fun onResume() {
        super.onResume()
        wv.onResume()
        if (pageTrusted) emit("lifecycle", JSONObject().put("state", "resume"))
    }

    override fun onRequestPermissionsResult(code: Int, p: Array<out String>, r: IntArray) {
        super.onRequestPermissionsResult(code, p, r)
        if (code == REQ_NOTIF) {
            val ok = r.isNotEmpty() && r[0] == PackageManager.PERMISSION_GRANTED
            emit("permission", JSONObject().put("name", "notifications").put("granted", ok))
        }
    }

    @Suppress("DEPRECATION")
    override fun onActivityResult(code: Int, result: Int, data: Intent?) {
        super.onActivityResult(code, result, data)
        if (code != REQ_NATIVE_GAME) return
        val id = pendingNativeId ?: return
        pendingNativeId = null
        val o = JSONObject().put("id", id)
        if (result == RESULT_OK && data != null) {
            val score = data.getIntExtra(NativeGames.EXTRA_SCORE, -1)
            val secs = data.getIntExtra(NativeGames.EXTRA_SECONDS, 0)
            o.put("status", "finished").put("score", if (score < 0) 0 else score).put("seconds", if (secs < 0) 0 else secs)
        } else {
            o.put("status", "cancelled")
        }
        emit("nativeGame", o)
    }

    override fun onSaveInstanceState(o: Bundle) { super.onSaveInstanceState(o); wv.saveState(o) }

    @Suppress("DEPRECATION")
    override fun onBackPressed() {
        wv.evaluateJavascript("(window.HB&&HB.back)?HB.back():false") { r ->
            if (r == "true") return@evaluateJavascript
            val now = System.currentTimeMillis()
            if (now - lastBack < 2000) finish()
            else { lastBack = now; Toast.makeText(this, "Apasă încă o dată pentru a ieși", Toast.LENGTH_SHORT).show() }
        }
    }

    override fun onDestroy() {
        pageTrusted = false
        try { wv.removeJavascriptInterface(HostBridge.NAME) } catch (e: Exception) {}
        try { (wv.parent as? ViewGroup)?.removeView(wv) } catch (e: Exception) {}
        try { wv.destroy() } catch (e: Exception) {}
        super.onDestroy()
    }
}

create or replace function public.handle_new_user() returns trigger
language plpgsql security definer set search_path = public as $$
declare
  h text := lower(nullif(trim(new.raw_user_meta_data->>'handle'), ''));
  n text := nullif(trim(new.raw_user_meta_data->>'username'), '');
begin
  if coalesce(new.is_anonymous, false) then return new; end if;
  if n is not null then n := left(n, 32); end if;

  end;
  return new;
end $$;

    order by s.score desc, s.updated_at asc, s.user_id limit 10) t;

    from public.game_scores s where s.game_id = p_game) r where r.user_id = uid;

package ro.lingurar.hiblue.bridge

import android.webkit.JavascriptInterface
import org.json.JSONArray
import org.json.JSONObject

/**
 * Acțiunile pe care host-ul le pune la dispoziția punții. Implementate de MainActivity.
 * Metodele pot fi apelate din firul JavaBridge (nu din UI); implementarea trebuie să sară pe UI când e nevoie.
 */
interface HostActions {
    /** Pagina încărcată acum în WebView este pagina locală a aplicației (assets)? */
    fun trustedPage(): Boolean
    fun vibrate(kind: Int)
    fun notificationsGranted(): Boolean
    fun requestNotifications()
    fun notify(title: String, body: String): Boolean
    fun setOrientation(mode: String): Boolean
    /** null = pornit, altfel codul erorii */
    fun launchNativeGame(id: String): String?
    /** (id, runtime) */
    fun nativeGames(): List<Pair<String, String>>
}

/**
 * Singura suprafață Kotlin expusă JavaScript-ului: window.HBNative.invoke(json) -> json.
 *
 * Cerere:   {"v":1,"method":"device.vibrate","params":{"kind":1}}
 * Răspuns:  {"v":1,"ok":true,"data":{...}}  sau  {"v":1,"ok":false,"error":{"code":"...","message":"..."}}
 *
 * Reguli: metode pe listă albă, parametri validați strict, cerere limitată ca dimensiune,
 * refuz dacă pagina curentă nu este cea locală. Nu se expun tokenuri, parole sau chei.
 */
class HostBridge(private val actions: HostActions, private val appVersion: String) {

    companion object {
        const val NAME = "HBNative"
        const val PROTOCOL = 1
        private const val MAX_REQUEST = 8 * 1024
        private val CAPABILITIES = listOf(
            "device.vibrate",
            "device.notifications",
            "device.notify",
            "host.orientation",
            "game.native"
        )
        private val GAME_ID = Regex("^[a-z0-9][a-z0-9_-]{1,31}$")
    }

    private fun ok(data: JSONObject): String =
        JSONObject().put("v", PROTOCOL).put("ok", true).put("data", data).toString()

    private fun fail(code: String, message: String): String =
        JSONObject().put("v", PROTOCOL).put("ok", false)
            .put("error", JSONObject().put("code", code).put("message", message)).toString()

    @JavascriptInterface
    fun invoke(request: String?): String {
        if (!actions.trustedPage()) return fail("FORBIDDEN", "Pagină neautorizată")
        if (request == null || request.length > MAX_REQUEST) return fail("BAD_REQUEST", "Cerere invalidă")
        val req: JSONObject = try {
            JSONObject(request)
        } catch (e: Exception) {
            return fail("BAD_REQUEST", "Cerere invalidă")
        }
        val method = req.optString("method", "")
        val params = req.optJSONObject("params") ?: JSONObject()
        return try {
            dispatch(method, params)
        } catch (e: Exception) {
            fail("INTERNAL", "Eroare internă")
        }
    }

    private fun dispatch(method: String, p: JSONObject): String {
        when (method) {
            "host.info" -> {
                return ok(
                    JSONObject().put("protocol", PROTOCOL).put("app", appVersion)
                        .put("capabilities", JSONArray(CAPABILITIES))
                )
            }
            "device.vibrate" -> {
                val k = p.opt("kind")
                if (k !is Int || k < 0 || k > 2) return fail("BAD_PARAMS", "kind trebuie să fie 0, 1 sau 2")
                actions.vibrate(k)
                return ok(JSONObject())
            }
            "device.notifications.status" -> {
                return ok(JSONObject().put("granted", actions.notificationsGranted()))
            }
            "device.notifications.request" -> {
                actions.requestNotifications()
                return ok(JSONObject().put("requested", true))
            }
            "device.notify" -> {
                val t = p.opt("title")
                val b = p.opt("body")
                if (t !is String || b !is String || t.isEmpty() || t.length > 60 || b.length > 200) {
                    return fail("BAD_PARAMS", "title (1-60) și body (0-200) sunt obligatorii")
                }
                if (!actions.notificationsGranted()) return fail("PERMISSION_DENIED", "Notificările nu sunt permise")
                return if (actions.notify(t, b)) ok(JSONObject()) else fail("INTERNAL", "Notificarea nu a putut fi afișată")
            }
            "host.orientation" -> {
                val m = p.opt("mode")
                if (m !is String || (m != "portrait" && m != "landscape" && m != "auto")) {
                    return fail("BAD_PARAMS", "mode: portrait | landscape | auto")
                }
                return if (actions.setOrientation(m)) ok(JSONObject()) else fail("INTERNAL", "Orientare neschimbată")
            }
            "game.native.list" -> {
                val arr = JSONArray()
                for (g in actions.nativeGames()) arr.put(JSONObject().put("id", g.first).put("runtime", g.second))
                return ok(JSONObject().put("games", arr))
            }
            "game.native.launch" -> {
                val id = p.opt("id")
                if (id !is String || !GAME_ID.matches(id)) return fail("BAD_PARAMS", "id de joc invalid")
                val err = actions.launchNativeGame(id)
                return if (err == null) ok(JSONObject().put("launched", true)) else fail(err, "Jocul nativ nu a putut porni")
            }
            else -> return fail("UNKNOWN_METHOD", "Metodă necunoscută")
        }
    }
}

    }

    @Suppress("DEPRECATION")
    fun vibrate(kind: Int) {
        try {
            val v = ctx.getSystemService(Context.VIBRATOR_SERVICE) as Vibrator
            if (!v.hasVibrator()) return
            if (Build.VERSION.SDK_INT >= 29) {
                val e = when (kind) {
                    0 -> VibrationEffect.EFFECT_TICK
                    1 -> VibrationEffect.EFFECT_CLICK
                    else -> VibrationEffect.EFFECT_HEAVY_CLICK
                }
                v.vibrate(VibrationEffect.createPredefined(e))
            } else {
                val ms = when (kind) {
                    0 -> 12L
                    1 -> 20L
                    else -> 32L
                }
                if (Build.VERSION.SDK_INT >= 26) {
                    v.vibrate(VibrationEffect.createOneShot(ms, VibrationEffect.DEFAULT_AMPLITUDE))
                } else {
                    v.vibrate(ms)
                }
            }
        } catch (e: Exception) {
        }
    }

    fun notificationsGranted(): Boolean =
        Build.VERSION.SDK_INT < 33 ||
            ctx.checkSelfPermission(Manifest.permission.POST_NOTIFICATIONS) == PackageManager.PERMISSION_GRANTED

    @Suppress("DEPRECATION")
    fun notify(title: String, body: String): Boolean {
        if (!notificationsGranted()) return false
        return try {
            val nm = ctx.getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
            val builder: Notification.Builder
            if (Build.VERSION.SDK_INT >= 26) {

                builder = Notification.Builder(ctx)

            )

                .setContentText(body)

# HiBlue · Android Host și NativeBridge

## Rolul Android Host
Android (Kotlin) este stratul nativ de bază. Interfața rămâne în `assets/` (HTML/CSS/JS).

| Responsabilitate | Fișier |
|---|---|
| Lifecycle, WebView, inset-uri, Back, permisiuni | `MainActivity.kt` |
| Puntea JS → Kotlin (listă albă de metode, validare) | `bridge/HostBridge.kt` |
| Vibrații, notificări | `host/DeviceServices.kt` |
| Registru generic pentru jocuri native | `nativegames/NativeGames.kt` |
| Client JS al punții | `assets/js/native-bridge.js` |

## Protocol
JS apelează `window.HBNative.invoke(json)` (prin `NativeBridge.call`). Metodele disponibile includ `host.info`, `device.vibrate`, `device.notifications.status`, `device.notifications.request`, `device.notify`, `host.orientation`, `game.native.list` și `game.native.launch`.

Evenimentele Kotlin → JS includ lifecycle, permisiuni și rezultatul unui joc nativ. Registrul este gol în versiunea 0.0.12; infrastructura este păstrată pentru eventuale jocuri native viitoare.

## Securitate
- Se expune o singură metodă (`invoke`); nu există tokenuri, parole sau chei în punte.
- Puntea refuză apelurile dacă pagina curentă nu provine din `file:///android_asset/...`.
- Navigarea WebView către adrese externe este interceptată; linkurile acceptate se deschid în browserul sistemului.
- WebView nu primește acces general la fișiere externe sau geolocație.

## Originea locală și compatibilitatea
Aplicația continuă să folosească `file:///android_asset/` pentru a păstra originea WebView și datele locale existente ale instalărilor. O schimbare de origine ar necesita o migrare explicită a datelor locale.

    error: ["destroyed"],

# Cum adaugi un joc Web nou

1. **Backend (o dată pe joc):** în Supabase SQL Editor:
   `insert into public.games (id,name,max_score,pace) values ('mygame','My Game',9999,0.5) on conflict (id) do update set name=excluded.name, max_score=excluded.max_score, pace=excluded.pace;`
   (`pace` = secunde minime per punct; respinge scoruri imposibile.) Fără rând în `games`, scorurile/progresul sunt respinse.
2. **Manifest:** adaugă în `app/src/main/assets/games/catalog.js` un obiect:
   `{id:"mygame", name:"My Game", version:"1.0.0", type:"arcade", runtime:"web", entry:"games/mygame.js", orientation:"portrait", capabilities:["identity","scores","leaderboard","haptics","events"], meta:{tagline:"...", developer:"...", description:"..."}, launch:{continueAfterScore:false}}`
   Manifestul invalid este refuzat și raportat în consolă (`Manifest invalid (...)`).
3. **Scriptul jocului** `app/src/main/assets/games/mygame.js`:
   ```js
   (function(){
     var GP = window.GamePlatform;
     function MyGame(host, api){           // host = element gol, pe tot ecranul
       // desenează în host; api.sdk.* pentru progres/scoruri; api.over(score, secs) la final
       return { restart(){}, pause(){}, resume(){}, destroy(){ /* elimină ORICE listener/timer/element */ } };
     }
     GP.registry.registerWeb("mygame", MyGame, { cover: function(k){ return '<svg viewBox="0 0 300 400">...</svg>'; } });
   })();
   ```
   `destroy()` este obligatoriu și trebuie să elibereze tot. `pause/resume/restart` sunt opționale.
4. **Încărcare:** adaugă `<script src="games/mygame.js"></script>` în `index.html` înainte de `platform/boot.js`
   (sau lasă platforma să-l încarce la Play din `entry`, dacă nu e inclus static).
5. **Progres:** `await api.sdk.progress.load()` la start, `api.sdk.progress.save({...})` la puncte de control.
   Nu folosi `localStorage` / Supabase direct și nu implementa autentificare.
6. Pentru orientare `landscape`/`auto` setează `orientation` în manifest; Launcher o aplică și o restaurează la ieșire.

Jocurile vechi pot continua să folosească `window.GM.games.push({...})` (compatibilitate păstrată).

# HiBlue · Arhitectură (bază 0.0.12)

HiBlue este o aplicație hibridă: **Android nativ** ca host, **HTML/CSS/JavaScript** pentru interfață, **GamePlatform** pentru jocuri și **Supabase** ca backend comun.

## Straturi

- **Android Host:** `MainActivity.kt`, lifecycle, WebView, permisiuni și puntea controlată către servicii native.
- **HostBridge:** listă albă de metode și validarea apelurilor JS → Kotlin.
- **DeviceServices:** vibrații și notificări.
- **GamePlatform:** Catalog, Registry, Runtime, Launcher, adaptoare, SDK, Storage și backend.
- **Jocuri web active:** Flyby (`games/flyby.js`) și Flip Nova (`game_flipnova.js`).
- **Supabase:** autentificare, profiluri, scoruri, ratinguri și progres sincronizat prin API-urile dedicate.

## Fluxul lansării unui joc

UI (`Play`) → `Launcher.launch(id)` → `Catalog.get(id)` → adaptorul indicat de manifest → încărcare și pornire → SDK. La ieșire, adaptorul este oprit și resursele sesiunii sunt eliberate.

## Reguli de întreținere

- Jocurile web noi folosesc GamePlatform și SDK-ul comun; nu accesează direct Supabase.
- Manifestele jocurilor sunt păstrate în `app/src/main/assets/games/catalog.js`.
- Scripturile jocurilor se încarcă numai dacă sunt referite de manifest sau de ordinea explicită din `index.html`.
- Adaptorul nativ este infrastructură generică; versiunea 0.0.12 nu înregistrează niciun joc nativ.
- Datele jocurilor eliminate nu trebuie să lase importuri, manifestări Android, dependențe Gradle sau scripturi active.

# Shared Game SDK (v1)

Contract comun pentru orice joc. Jocul primește `sdk` (web: `api.sdk`); nu primește tokenuri, parole, chei sau acces Android.
Toate metodele întorc Promise; erorile au `error.code` din: `BAD_PARAMS`, `CAPABILITY_DENIED`, `SESSION_INVALID`,
`RATE_LIMITED`, `FORBIDDEN`, `NOT_FOUND`, `OFFLINE`, `UNKNOWN_METHOD`, `BAD_REQUEST`, `INTERNAL`.

| API | Capabilitate | Descriere |
|---|---|---|
| `sdk.identity.get()` | identity | `{kind:"user"\|"guest", id, name, handle}` |
| `sdk.progress.load(slot?)` / `save(data, slot?)` | progress | obiect/listă JSON ≤ 64 KB; local-first, sincronizat cu Supabase; `save` -> `{saved, synced}` |
| `sdk.scores.submit(score, secs)` | scores | trimite scorul; `{submitted, record, local_best, in_top, rank, best, min, improved}` sau `{submitted:false, reason:"guest"\|"offline"\|"rejected"}` |
| `sdk.scores.best()` | scores | cel mai bun scor local |
| `sdk.leaderboard.top(limit?)` | leaderboard | `[{pos,name,handle,score,me}]` (max 10; necesită cont) |
| `sdk.sync.flush()` | sync | retrimite progres și scoruri în așteptare -> `{pushed, pending}` |
| `sdk.haptics.vibrate(0\|1\|2)` | haptics | respectă setarea „Haptics” |
| `sdk.audio.tone(freq, ms, vol?)` | audio | ton WebAudio (nu există audio nativ în host) |
| `sdk.events.emit(name, data?)` | events | eveniment de joc (nume `^[a-z][a-z0-9_.]{0,31}$`, date ≤ 2 KB) |
| `sdk.session.end(reason?)`, `onPause/onResume/onExit(fn)` | — | închiderea și reluarea sesiunii |

## Protocol de mesaje (pentru transporturi non-JS)
Cerere: `{"v":1,"id":1,"session":"<token>","game":"flyby","method":"scores.submit","params":{"score":12,"secs":30}}`
Răspuns: `{"v":1,"id":1,"ok":true,"data":{...}}` sau `{"v":1,"id":1,"ok":false,"error":{"code":"...","message":"..."}}`.
Verificări, în ordine: versiune -> origine permisă (`internal`, `native-host`; altele doar prin `allowOrigin`) -> sesiune activă
(token) -> joc = jocul sesiunii -> metodă pe listă albă -> capabilitate din manifest -> parametri -> limită de rată.

## Securitate și încredere
- Scorul din client NU este de încredere: `submit_score` (server) respinge scoruri peste `max_score` sau peste ritmul
  `pace` al jocului și păstrează doar Top 10. Este o verificare de plauzibilitate, nu anti-cheat criptografic.
- Progresul se scrie doar prin `game_progress_put` (RLS: fiecare utilizator își vede doar rândurile).
- Nu se folosește `service_role` nicăieri în client.
- Jocurile web de încredere rulează în același context JS cu aplicația: SDK-ul limitează *contractul*, nu poate împiedica
  cod malițios din același context. Jocuri neîncredere ar cere un iframe sandbox (necesită schimbarea CSP `frame-src`).

## Straturi
API expus jocului: `platform/sdk.js` · Implementare Web/host: serviciile din `platform/boot.js` ·
Android: `NativeBridge` + `HostBridge.kt` · Backend: `platform/backend.js` + `supabase/update0.0.7.sql`.

# Integrarea unui joc Native (Kotlin / Godot / Unity)

**Stare:** infrastructura există și este testată doar la nivel de contract JS; **nu există niciun joc nativ** și
lansarea nu a fost rulată pe dispozitiv. Registrul din host este gol.

## Pași
1. **Activity-ul jocului** (modul Kotlin, export Godot sau Unity) declarat în `AndroidManifest.xml`.
2. **Înregistrare în host:** în `nativegames/NativeGames.kt`, în `registerBuiltIn()`:
   `register(NativeGameEntry("mygame", "kotlin") { ctx -> Intent(ctx, MyGameActivity::class.java) })`
   (`runtime`: `kotlin` | `godot` | `unity`; id `^[a-z0-9][a-z0-9_-]{1,31}$`).
3. **Manifest** în `games/catalog.js`: `runtime:"native-kotlin"` (sau `native-godot` / `native-unity`), `entry:"mygame"` (id-ul de la pasul 2),
   `capabilities:["scores", ...]`. Rând în `public.games` (vezi ADDING_WEB_GAME.md, pasul 1).
4. **Rezultat:** la final, Activity-ul face `setResult(RESULT_OK, Intent().putExtra(NativeGames.EXTRA_SCORE, score).putExtra(NativeGames.EXTRA_SECONDS, secs)); finish()`.
   Host-ul emite `nativeGame {id,status,score,seconds}`; Native Game Adapter trimite scorul prin SDK (revalidat pe server).
5. **Limitări actuale:** jocul nativ primește servicii doar prin rezultatul Intent (scor, secunde). Progresul și clasamentele
   în timpul sesiunii native nu sunt expuse încă (ar cere un serviciu legat/Binder; tokenurile NU trebuie date jocului nativ).
   Cover-ul din cameră este unul generic (litera jocului) până se adaugă o imagine în manifest/UI.
6. Godot/Unity necesită adăugarea bibliotecilor lor în Gradle: nu sunt incluse și nu sunt testate aici.

-- Rulează O SINGURĂ DATĂ în Supabase → SQL Editor, după update0.0.4.sql. Idempotent (poate fi rulat din nou).
--
-- Adaugă DOAR ce lipsește: tabela public.game_progress + 2 funcții RPC.
-- NU modifică profiles, games, game_scores, game_ratings și nici funcțiile existente (submit_score, game_top, ...).
-- Dacă nu rulezi această migrație, aplicația funcționează: progresul rămâne local pe dispozitiv
-- (SDK-ul raportează synced:false) și se sincronizează automat după ce migrația există.
--
-- Reguli:
--  * Un rând = progresul unui utilizator pentru un joc și un slot (ex. 'main'). Doar proprietarul îl vede.
--  * Scrierea se face EXCLUSIV prin game_progress_put() (clientul nu poate scrie direct în tabel).
--  * Validare pe server: joc existent în public.games, slot valid, max 64 KB / rând, max 8 sloturi / joc / utilizator.
--  * Concurență: versiune optimistă. Dacă base != versiunea serverului, nu se scrie și se întoarce
--    {"conflict":true,"version":N}; clientul decide (implicit: ultima scriere câștigă, retrimite cu noua bază).
--  * Nu se folosește service_role; funcțiile rulează cu drepturile definer doar pentru rândul auth.uid().
-- =====================================================================

create table if not exists public.game_progress (
  game_id    text not null references public.games(id) on delete cascade,
  user_id    uuid not null references auth.users(id) on delete cascade,
  slot       text not null check (slot ~ '^[a-z0-9_-]{1,24}$'),
  data       jsonb not null,
  version    int  not null default 1,
  updated_at timestamptz not null default now(),
  primary key (game_id, user_id, slot)
);
alter table public.game_progress enable row level security;
drop policy if exists gp_select on public.game_progress;
create policy gp_select on public.game_progress for select to authenticated using (user_id = auth.uid());
revoke insert, update, delete on public.game_progress from anon, authenticated;

-- citire: {data, version, updated_at} sau null
create or replace function public.game_progress_get(p_game text, p_slot text) returns json
language plpgsql stable security definer set search_path = public as $$
declare uid uuid := auth.uid(); r public.game_progress%rowtype;
begin
  if uid is null then raise exception 'forbidden' using errcode = '42501'; end if;
  select * into r from public.game_progress where game_id = p_game and user_id = uid and slot = p_slot;
  if not found then return null; end if;
  return json_build_object('data', r.data, 'version', r.version, 'updated_at', r.updated_at);
end $$;

-- scriere cu versiune optimistă: {"ok":true,"version":N} sau {"conflict":true,"version":N}
create or replace function public.game_progress_put(p_game text, p_slot text, p_data jsonb, p_base int default 0) returns json
language plpgsql security definer set search_path = public as $$
declare uid uuid := auth.uid(); r public.game_progress%rowtype; n int;
begin
  if uid is null then raise exception 'forbidden' using errcode = '42501'; end if;
  if not exists (select 1 from public.games where id = p_game) then raise exception 'joc necunoscut' using errcode = '22023'; end if;
  if p_slot is null or p_slot !~ '^[a-z0-9_-]{1,24}$' then raise exception 'slot invalid' using errcode = '22023'; end if;
  if p_data is null or jsonb_typeof(p_data) not in ('object', 'array') then raise exception 'date invalide' using errcode = '22023'; end if;
  if octet_length(p_data::text) > 65536 then raise exception 'progres prea mare' using errcode = '22023'; end if;

  perform pg_advisory_xact_lock(hashtext('progress:' || uid::text || ':' || p_game || ':' || p_slot));
  select * into r from public.game_progress where game_id = p_game and user_id = uid and slot = p_slot;
  if found then
    if coalesce(p_base, 0) <> r.version then
      return json_build_object('conflict', true, 'version', r.version);
    end if;
    update public.game_progress set data = p_data, version = r.version + 1, updated_at = now()
      where game_id = p_game and user_id = uid and slot = p_slot
      returning version into n;
    return json_build_object('ok', true, 'version', n);
  end if;

  if (select count(*) from public.game_progress where game_id = p_game and user_id = uid) >= 8 then
    raise exception 'prea multe sloturi' using errcode = '22023';
  end if;
  insert into public.game_progress (game_id, user_id, slot, data, version) values (p_game, uid, p_slot, p_data, 1);
  return json_build_object('ok', true, 'version', 1);
end $$;

revoke all on function public.game_progress_get(text, text)            from public;
revoke all on function public.game_progress_put(text, text, jsonb, int) from public;
grant execute on function public.game_progress_get(text, text)            to authenticated;
grant execute on function public.game_progress_put(text, text, jsonb, int) to authenticated;
