package ro.lingurar.hiblue.media

import android.content.ContentResolver
import android.content.Context
import android.net.Uri
import android.os.Build
import android.provider.OpenableColumns
import androidx.media3.common.MediaItem
import androidx.media3.common.Player
import androidx.media3.exoplayer.ExoPlayer
import androidx.media3.session.MediaSession

class MediaPlayerBridge(private val context: Context) {
    private var player: ExoPlayer? = null
    private var session: MediaSession? = null
    private var activeType: String = "audio"

    fun ensurePlayer(mode: String): ExoPlayer {
        if (player == null) {
            player = ExoPlayer.Builder(context).build()
            player?.playWhenReady = true
            player?.repeatMode = Player.REPEAT_MODE_OFF
        }
        activeType = mode
        return player!!
    }

    fun playFile(uri: Uri, mode: String, title: String?) {
        val p = ensurePlayer(mode)
        val item = MediaItem.Builder().setUri(uri).setMediaId(title ?: uri.toString()).build()
        p.setMediaItem(item)
        p.prepare()
        p.play()
    }

    fun playUrl(url: String, mode: String, title: String?) {
        val p = ensurePlayer(mode)
        val item = MediaItem.Builder().setUri(url).setMediaId(title ?: url).build()
        p.setMediaItem(item)
        p.prepare()
        p.play()
    }

    fun pause() = player?.pause()
    fun resume() = player?.play()
    fun stop() { player?.stop(); player?.clearMediaItems() }
    fun release() { player?.release(); player = null; session?.release(); session = null }
    fun currentPositionMs(): Long = player?.currentPosition ?: 0L
    fun durationMs(): Long = player?.duration ?: 0L
    fun seekTo(ms: Long) { player?.seekTo(ms) }
    fun setVolume(v: Float) { player?.volume = v }
    fun isPlaying(): Boolean = player?.isPlaying == true
    fun activeMode(): String = activeType

    fun resolveDisplayName(uri: Uri, resolver: ContentResolver): String {
        val name = resolver.query(uri, arrayOf(OpenableColumns.DISPLAY_NAME), null, null, null)?.use { c ->
            if (c.moveToFirst()) c.getString(c.getColumnIndexOrThrow(OpenableColumns.DISPLAY_NAME)) else null
        }
        return name ?: uri.lastPathSegment ?: "media"
    }

    companion object {
        fun supportsMedia3(): Boolean = true
        fun requestMediaPermissionsIfNeeded(context: Context) {
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
                // handled in app layer
            }
        }
    }
}

  const NativeBridge = window.NativeBridge || {};

      if (window.HBMedia && typeof window.HBMedia.openPicker === 'function') {

      }
      return { ok: false, reason: 'media picker unavailable' };

      if (window.HBMedia && mode === 'radio' && typeof window.HBMedia.playRadio === 'function') {
        window.HBMedia.playRadio(url, title || 'Radio');
        return { ok: true };
      }
      return { ok: false };
    }
  };
  window.NativeBridge = NativeBridge;

package ro.lingurar.hiblue

import android.app.Activity
import android.content.Intent
import android.net.Uri
import android.os.Bundle
import android.webkit.WebView
import android.webkit.WebViewClient
import android.webkit.WebChromeClient
import android.webkit.WebResourceRequest

class MainActivity : Activity() {
    private lateinit var wv: WebView

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        wv = WebView(this)
        setContentView(wv)
        wv.settings.javaScriptEnabled = true
        wv.settings.domStorageEnabled = true
        wv.settings.allowFileAccess = true
        wv.webChromeClient = WebChromeClient()
        wv.webViewClient = object : WebViewClient() {
            override fun shouldOverrideUrlLoading(view: WebView?, request: WebResourceRequest?): Boolean = false
        }
        wv.loadUrl("file:///android_asset/index.html")
    }

    private fun pickMedia(mode: String) {
        val mime = if (mode == "video") "video/*" else "audio/*"
        val intent = Intent(Intent.ACTION_OPEN_DOCUMENT).apply {
            addCategory(Intent.CATEGORY_OPENABLE)
            type = mime
            putExtra(Intent.EXTRA_MIME_TYPES, arrayOf(mime))
        }
        startActivityForResult(intent, if (mode == "video") 1001 else 1002)
    }

    override fun onActivityResult(requestCode: Int, resultCode: Int, data: Intent?) {
        super.onActivityResult(requestCode, resultCode, data)
        if (resultCode != RESULT_OK || data == null || data.data == null) return
        val uri = data.data!!
        val mode = if (requestCode == 1001) "video" else "audio"
        val js = """
        (function(){
          if (window.HBMedia && window.HBMedia.state) {
            window.HBMedia.state.current = { title: '${uri.lastPathSegment ?: "media"}', mode: '$mode', url: '${uri}', radio: false, duration: 0, position: 0 };
            window.HBMedia.state.playing = true;
            window.HBMedia.ensure && window.HBMedia.ensure();
          }
        })();
        """.trimIndent()
        wv.evaluateJavascript(js, null)
    }
}
