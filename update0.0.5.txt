# HiBlue – update0.0.5

**My Blue** primul update major de grafică, fizică și conținut. Ursulețul a fost redesenat
complet – mai moale, mai rotund, cu ochi mari și sclipitori, obraji roz, expresii care se
schimbă după dispoziție, clipire, respirație și mișcări de urechi. Dacă îl atingi, reacționează:
sare, îl cuprinde un nor de inimioare, obrajii îi rumenesc și primește bucurie.

## Ce e nou în My Blue 0.0.5

- **Ursuleț nou, drăguț și expresiv** – proporții mai rotunde, blană cu gradient cald,
  ochi sclipitori cu reflexii, obraji roz, cusături de jucărie de pluș, bot mobil.
- **Dispoziție live** – Blue are patru stări (fericit, somnoros, flămând, murdar) care îi
  schimbă ochii, gura și aura. Când e fericit lasă scântei, când e somnoros apare un „Z”,
  când îi e foame apare o broboană, când e murdar are nori de praf.
- **Atinge-l ca să-l mângâi** – un tap pe Blue îi dă squash & stretch, inimioare zburătoare,
  vibrație, +bucurie și obraji rumeni. Dublu-tap = îmbrățișare și rotire.
- **Trage-l prin cameră** – în modul casă poți muta ursulețul cu degetul unde vrei.
- **Fizică nouă** – respirație sinusoidală, clipire periodică, urechi care se mișcă,
  bounce la aterizare, idle bobbing, coadă care se leagănă.
- **12 haine/skinuri noi** – rochie de prințesă, smoching, hanorac, jachetă denim,
  pijama cu stele, costum de baie, kimono, aripi de zână, haină de detectiv, echipament de
  brutar, fular de pilot, cizme de ploaie. Fiecare e desenată detaliat pe ursuleț.
- **Camere mai bogate** – lumină caldă, umbre moi, ferestre cu cer animat, obiecte care
  pulsează ușor.
- **Performanță** – animațiile folosesc transformări canvas, nu recreăm obiecte, deci
  merg fluent și pe telefoane mai slabe.

## Aplicare

Pune `update0.0.5.txt` în rădăcina repo-ului alături de `hiblue.txt` și update-urile
anterioare (dacă le ai separate), apoi construiește APK-ul cu workflow-ul `Build APK`.
Salvarea locală a lui Blue (`hiblue.myblue.save.v1`) rămâne compatibilă – monedele,
ținutele și decorul existente se păstrează. Noile haine apar ca nebăgate în dulap
(le poți cumpăra din Games → My Blue → Haine).

## Supabase

Nicio schimbare de schemă în 0.0.5 – `myblue` există deja în `public.games` din
`update0.0.4.sql`, scorurile de quest continuă să meargă spre Top 10.

## Controale

- **Casă**: tap pe Blue = mângâiere; dublu-tap = îmbrățișare; drag = mută ursulețul;
  butoanele de jos pentru hrănit/baie/joacă/somn.
- **Quest**: joystick stânga-jos, săgețile, WASD sau tap pe hartă ca destinație.

    function moveQuest(dt) {
      const q = activeQuest;
      if (!q) return;
      let ix = joy.x, iy = joy.y;
      if (keys.has("ArrowLeft") || keys.has("KeyA")) ix -= 1;
      if (keys.has("ArrowRight") || keys.has("KeyD")) ix += 1;
      if (keys.has("ArrowUp") || keys.has("KeyW")) iy -= 1;
      if (keys.has("ArrowDown") || keys.has("KeyS")) iy += 1;
      if (mapTarget) {
        const dx = mapTarget.x - q.x * W, dy = mapTarget.y - q.y * H;
        if (Math.hypot(dx, dy) < 15) mapTarget = null;
        else { ix = dx; iy = dy; }
      }
      const len = Math.hypot(ix, iy);
      if (len > .02) {
        ix /= len; iy /= len;
        const speed = q.def.vehicle ? 122 : 148;
        q.x = clamp(q.x + ix * speed * dt / W, .06, .94);
        q.y = clamp(q.y + iy * speed * dt / H, .12, .91);
        if (q.def.vehicle) {
          if (q.x < .2) q.y = clamp(q.y, .53, .91);
          else if (q.x > .77) q.y = clamp(q.y, .12, .57);
          else q.y = clamp(q.y, .43, .65);
        }
        q.angle = Math.atan2(iy, ix);
      }
      if (q.def.vehicle) {
        if (q.leg === "toStore") {
          const store = worldPoint(.83, .22);
          if (Math.hypot(q.x * W - store.x, q.y * H - store.y) < Math.min(W, H) * .12) {
            q.leg = "shopping"; say("Ai ajuns! Pune cele trei alimente în coș.");
          }
        } else if (q.leg === "shopping") {
          q.items.forEach((it) => {
            if (!it.done && Math.hypot(q.x - it.x, q.y - it.y) < .095) collect(it);
          });
        } else if (q.leg === "toHome") {
          const home = worldPoint(.14, .81);
          if (Math.hypot(q.x * W - home.x, q.y * H - home.y) < Math.min(W, H) * .12) completeQuest();
        }
      } else {
        q.items.forEach((it) => {
          if (!it.done && Math.hypot((q.x - it.x) * W, (q.y - it.y) * H) < Math.min(W, H) * .075) collect(it);
        });
      }
      const oldEnergy = Math.round(state.stats.energy);
      state.stats.energy = clamp(state.stats.energy - dt * .12, 0, 100);
      if (Math.round(state.stats.energy) !== oldEnergy) dirtyStats = true;
    }

    function roundRect(x, y, w, h, r, fill, stroke, lw) {
      r = Math.min(r, w / 2, h / 2);
      c.beginPath(); c.moveTo(x + r, y); c.arcTo(x + w, y, x + w, y + h, r);
      c.arcTo(x + w, y + h, x, y + h, r); c.arcTo(x, y + h, x, y, r);
      c.arcTo(x, y, x + w, y, r); c.closePath();
      if (fill) { c.fillStyle = fill; c.fill(); }
      if (stroke) { c.strokeStyle = stroke; c.lineWidth = lw || 1; c.stroke(); }
    }
    function ellipse(x, y, rx, ry, fill, stroke, lw) {

      if (fill) { c.fillStyle = fill; c.fill(); }
      if (stroke) { c.strokeStyle = stroke; c.lineWidth = lw || 1; c.stroke(); }
    }

    function frame(now) {
      if (destroyed) return;
      raf = requestAnimationFrame(frame);
      const dt = Math.min(.04, Math.max(0, (now - (lastFrame || now)) / 1000));
      lastFrame = now; elapsed += dt;

      // animațiiBlue (indiferent de mod)
      breathing += dt;
      earWiggle += dt;
      idleBob += dt;
      blinkTimer -= dt;
      if (blinkTimer <= 0 && blinkPhase <= 0) {
        // la sleepy, blink mai frecvent și mai lung (ca și cum moțăie)
        if (moodState === "sleepy") {
          blinkPhase = 0.5 + Math.random() * 0.4;
          blinkTimer = 1.2 + Math.random() * 1.8;
        } else {
          blinkPhase = 0.14;
          blinkTimer = 2.5 + Math.random() * 3.5;
        }
      }
      if (blinkPhase > 0) blinkPhase -= dt;

      // squash relaxează înapoi spre 1
      blueSquash.x = lerp(blueSquash.x, 1, Math.min(1, dt * 8));
      blueSquash.y = lerp(blueSquash.y, 1, Math.min(1, dt * 8));
      // blush scade
      moodBlush = Math.max(0, moodBlush - dt * 0.6);

      // privit aleatoriu (idle)
      lookTimer -= dt;
      if (lookTimer <= 0) {
        lookingDir = Math.random() < 0.4 ? (Math.random() < 0.5 ? -1 : 1) : 0;
        lookTimer = 2 + Math.random() * 4;
      }

      // drag physică – Blue urmărește degetul cu inerție
      if (dragState.active && mode === "home") {
        // poziția target e trasă în pixeli, convertim în normalizat
        const tx = clamp(dragState.ox / W, 0.18, 0.82);
        const ty = clamp(dragState.oy / H, 0.45, 0.85);
        bluePos.x = lerp(bluePos.x, tx, Math.min(1, dt * 12));
        bluePos.y = lerp(bluePos.y, ty, Math.min(1, dt * 12));
      } else if (mode === "home") {
        // ușor revenire la centrul camerei dacă a fost mutat
        // (dacă vrei să rămână unde l-ai pus, comentează rândul următor)
        // bluePos.x = lerp(bluePos.x, 0.5, Math.min(1, dt * 1.2));
        // bluePos.y = lerp(bluePos.y, 0.69, Math.min(1, dt * 1.2));
      }

      if (mode === "quest" && activeQuest) {
        moveQuest(dt);
        if (activeQuest) activeQuest.spark = Math.max(0, activeQuest.spark - dt);
      }

      updateParticles(dt);

      c.setTransform(dpr,0,0,dpr,0,0);
      if (mode === "quest") drawQuest(elapsed);
      else drawHome(elapsed);

      // particule peste tot
      particles.forEach(drawParticle);

      if (dirtyStats) { updateHUD(); dirtyStats = false; }
    }

    /* ===== Input: tap pe Blue, drag ===== */
    function hitTestBlue(px, py) {
      const bx = bluePos.x * W, by = bluePos.y * H;
      const size = Math.min(W*.55,H*.62,220);
      const r = size * 0.5;
      return Math.hypot(px - bx, py - by + 20) < r;
    }
    function stageDown(e) {
      if (mode !== "home") {
        // în quest, tap = setează destinație (dacă nu lovește joystick)
        if (mode === "quest" && !e.target.closest("[data-joystick]")) {
          const r = canvas.getBoundingClientRect();
          mapTarget = { x: e.clientX - r.left, y: e.clientY - r.top };
        }
        return;
      }
      const r = canvas.getBoundingClientRect();
      const px = e.clientX - r.left, py = e.clientY - r.top;
      if (hitTestBlue(px, py)) {
        dragState.active = true;
        dragState.pointer = e.pointerId;
        dragState.ox = px;
        dragState.oy = py;
        dragState.startT = performance.now();
        dragState.moved = false;
        canvas.setPointerCapture?.(e.pointerId);
      }
    }
    function stageMove(e) {
      if (!dragState.active || e.pointerId !== dragState.pointer) return;
      const r = canvas.getBoundingClientRect();
      const px = e.clientX - r.left, py = e.clientY - r.top;
      const dx = px - dragState.ox, dy = py - dragState.oy;
      if (Math.hypot(dx, dy) > 8) dragState.moved = true;
      dragState.ox = px; dragState.oy = py;
    }
    function stageUp(e) {
      if (!dragState.active || e.pointerId !== dragState.pointer) return;
      const wasMoved = dragState.moved;
      const elapsedTap = performance.now() - dragState.startT;
      dragState.active = false;
      dragState.pointer = null;
      // dacă a fost un tap scurt pe Blue (fără mișcare) = mângâiere
      if (!wasMoved && elapsedTap < 320) {
        petBlue();
      }
    }

    function onClick(e) {
      const t = e.target.closest("button");
      if (!t) return;
      if (t.dataset.view) return setView(t.dataset.view);
      if (t.dataset.room) return setRoom(t.dataset.room);
      if (t.dataset.action) return act(t.dataset.action);
      if (t.dataset.food) return buyFood(t.dataset.food);
      if (t.dataset.cloth) return buyCloth(t.dataset.cloth);
      if (t.dataset.decor) return buyDecor(t.dataset.decor);
      if (t.dataset.filter) { filter = t.dataset.filter; return renderPanel(); }
      if (t.hasAttribute("data-careback")) { careMode = "actions"; return renderPanel(); }
      if (t.dataset.startquest) return startQuest(t.dataset.startquest);
      if (t.hasAttribute("data-cancelquest")) return cancelQuest();
    }
    function keyDown(e) {
      const code = e.code || e.key;
      if (["ArrowUp","ArrowDown","ArrowLeft","ArrowRight","Space"].includes(code)) e.preventDefault();
      keys.add(code);
    }
    function keyUp(e) { keys.delete(e.code || e.key); }
    function joyDown(e) {
      e.preventDefault(); joystick.setPointerCapture?.(e.pointerId);
      const r = joystick.getBoundingClientRect();
      joy.pointer=e.pointerId;joy.active=true;joy.centerX=r.left+r.width/2;joy.centerY=r.top+r.height/2;
      joyMove(e);
    }
    function joyMove(e) {
      if (!joy.active || e.pointerId !== joy.pointer) return;
      const dx=e.clientX-joy.centerX,dy=e.clientY-joy.centerY,len=Math.hypot(dx,dy)||1,cap=34;
      const ratio=Math.min(1,cap/len);joy.x=dx/len*ratio;joy.y=dy/len*ratio;
      knob.style.left=`calc(50% + ${joy.x*34}px)`;knob.style.top=`calc(50% + ${joy.y*34}px)`;
    }
    function joyUp(e) {
      if(e.pointerId!==joy.pointer)return;
      joy.active=false;joy.pointer=null;joy.x=0;joy.y=0;knob.style.left="50%";knob.style.top="50%";
    }
    function pageHide() { save(); }

    root.addEventListener("click",onClick);
    window.addEventListener("keydown",keyDown,{passive:false});
    window.addEventListener("keyup",keyUp);
    window.addEventListener("pagehide",pageHide);
    window.addEventListener("resize",resize);
    canvas.addEventListener("pointerdown",stageDown);
    canvas.addEventListener("pointermove",stageMove);
    canvas.addEventListener("pointerup",stageUp);
    canvas.addEventListener("pointercancel",stageUp);
    joystick.addEventListener("pointerdown",joyDown);
    joystick.addEventListener("pointermove",joyMove);
    joystick.addEventListener("pointerup",joyUp);
    joystick.addEventListener("pointercancel",joyUp);
    if ("ResizeObserver" in window) {
      const observer = new ResizeObserver(resize);
      observer.observe(stage);
      root._mbObserver = observer;
    }
    applyOfflineTime(); resize(); updateHUD(); renderPanel();
    lastFrame=performance.now();raf=requestAnimationFrame(frame);
    return {
      restart() {
        if(activeQuest){activeQuest=null;mode="home";}
        view="home";room=ROOMS.some((r)=>r.id===state.room)?state.room:"living";
        bluePos={x:0.5,y:0.69};joy.x=0;joy.y=0;save();renderPanel();updateHUD();
      },
      destroy() {
        destroyed=true;cancelAnimationFrame(raf);clearTimeout(toastTimer);clearTimeout(bubbleTimer);
        root.removeEventListener("click",onClick);
        window.removeEventListener("keydown",keyDown);window.removeEventListener("keyup",keyUp);
        window.removeEventListener("pagehide",pageHide);window.removeEventListener("resize",resize);
        canvas.removeEventListener("pointerdown",stageDown);canvas.removeEventListener("pointermove",stageMove);
        canvas.removeEventListener("pointerup",stageUp);canvas.removeEventListener("pointercancel",stageUp);
        joystick.removeEventListener("pointerdown",joyDown);joystick.removeEventListener("pointermove",joyMove);
        joystick.removeEventListener("pointerup",joyUp);joystick.removeEventListener("pointercancel",joyUp);
        root._mbObserver?.disconnect();save();root.remove();css.remove();
      }
    };
  }
